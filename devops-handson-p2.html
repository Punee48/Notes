# ── backend.tf — Remote State Configuration ─────────────────────
# Remote state: store Terraform state in S3 instead of local.
# WHY: Team members share state, state survives laptop failures,
# enables state locking to prevent concurrent apply conflicts.

terraform {
  # required_providers = declare which providers and their versions.
  # ~> 5.0 = "compatible with 5.x" (allows 5.1, 5.2 but NOT 6.0)
  # Pinning versions ensures everyone on team uses same provider.
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
    kubernetes = {
      source  = "hashicorp/kubernetes"
      version = "~> 2.0"
    }
    helm = {
      source  = "hashicorp/helm"
      version = "~> 2.0"
    }
    tls = {
      source  = "hashicorp/tls"
      version = "~> 4.0"
    }
  }
  required_version = ">= 1.5.0"

  # backend "s3" = store terraform.tfstate in S3 bucket.
  # Before this works, create the bucket and DynamoDB table manually (chicken-egg!)
  # OR use: terraform init -backend=false, apply just the bucket, then migrate state.
  backend "s3" {
    bucket         = "fluxcart-terraform-state"
    key            = "prod/eks/terraform.tfstate"  # Path inside the bucket
    region         = "ap-south-1"
    # encrypt = true: encrypt state file at rest using AES-256.
    # State file contains SENSITIVE DATA (passwords, IPs, resource IDs)!
    encrypt        = true
    # dynamodb_table: used for STATE LOCKING.
    # If two people run terraform apply simultaneously, the second one WAITS.
    # DynamoDB table must have LockID as primary key (String type).
    dynamodb_table = "fluxcart-terraform-locks"
  }
}

# ═══════════════════════════════════════════════════════════════
# VPC (Virtual Private Cloud)
# Your private network on AWS. Like having your own data center.
# CIDR 10.0.0.0/16 = 65,536 IP addresses in this network.
# ═══════════════════════════════════════════════════════════════

resource "aws_vpc" "main" {
  cidr_block = var.vpc_cidr    # e.g., "10.0.0.0/16"

  # enable_dns_hostnames: assign DNS names to EC2 instances.
  # REQUIRED for EKS! Nodes use DNS for cluster communication.
  enable_dns_hostnames = true
  enable_dns_support   = true

  tags = {
    Name = "${var.project}-${var.environment}-vpc"
    # These special EKS tags tell the AWS Load Balancer Controller
    # which VPC to use when creating load balancers for Ingress resources.
    "kubernetes.io/cluster/${var.cluster_name}" = "shared"
  }
}

# Availability Zones — fetch dynamically, not hardcoded
data "aws_availability_zones" "available" {
  state = "available"
}

# PUBLIC SUBNETS — for load balancers, NAT gateways
# count.index = 0,1,2 → creates 3 public subnets in 3 AZs
resource "aws_subnet" "public" {
  count = length(var.public_subnet_cidrs)

  vpc_id     = aws_vpc.main.id
  # cidrsubnet("10.0.0.0/16", 8, 0) = "10.0.0.0/24" (256 IPs)
  # cidrsubnet("10.0.0.0/16", 8, 1) = "10.0.1.0/24"
  cidr_block = var.public_subnet_cidrs[count.index]

  # Place each subnet in a different AZ for high availability
  availability_zone = data.aws_availability_zones.available.names[count.index]

  # map_public_ip_on_launch: instances in this subnet get a public IP automatically.
  # Only for public subnets! Private subnets must never have this.
  map_public_ip_on_launch = true

  tags = {
    Name = "${var.project}-public-${count.index}"
    # This tag is REQUIRED for AWS Load Balancer Controller to place ALBs here
    "kubernetes.io/role/elb"                    = "1"
    "kubernetes.io/cluster/${var.cluster_name}" = "shared"
  }
}

# PRIVATE SUBNETS — for EKS worker nodes
resource "aws_subnet" "private" {
  count = length(var.private_subnet_cidrs)

  vpc_id            = aws_vpc.main.id
  cidr_block        = var.private_subnet_cidrs[count.index]
  availability_zone = data.aws_availability_zones.available.names[count.index]

  tags = {
    Name = "${var.project}-private-${count.index}"
    # internal-elb tag: ALB controller uses internal LBs in private subnets
    "kubernetes.io/role/internal-elb"           = "1"
    "kubernetes.io/cluster/${var.cluster_name}" = "shared"
  }
}

# INTERNET GATEWAY — connects the VPC to the public internet
resource "aws_internet_gateway" "main" {
  vpc_id = aws_vpc.main.id
  tags   = { Name = "${var.project}-igw" }
}

# ELASTIC IPs for NAT Gateways (static public IPs)
resource "aws_eip" "nat" {
  # count = 1 for cost savings (1 NAT Gateway), or = length(subnets) for HA
  count  = 1
  domain = "vpc"
  depends_on = [aws_internet_gateway.main]
  # depends_on: explicit dependency. Terraform waits for IGW before creating EIP.
}

# NAT GATEWAY — allows private subnet instances to access internet (outbound only)
# Private EC2/EKS nodes need internet for: pulling Docker images, OS updates, AWS API calls.
# NAT translates private IPs to the EIP public IP.
# Traffic flow: Private node → NAT GW → Internet Gateway → Internet
resource "aws_nat_gateway" "main" {
  count         = 1
  allocation_id = aws_eip.nat[0].id
  # NAT Gateway must be in a PUBLIC subnet (it needs an internet route)
  subnet_id     = aws_subnet.public[0].id
  depends_on    = [aws_internet_gateway.main]
}

# ROUTE TABLE for public subnets — route all traffic to IGW
resource "aws_route_table" "public" {
  vpc_id = aws_vpc.main.id
  route {
    # 0.0.0.0/0 = default route (all traffic not matching a more specific route)
    # gateway_id = send this traffic to the Internet Gateway
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.main.id
  }
}

# ROUTE TABLE for private subnets — route internet traffic to NAT GW
resource "aws_route_table" "private" {
  vpc_id = aws_vpc.main.id
  route {
    cidr_block     = "0.0.0.0/0"
    # nat_gateway_id = internet-bound traffic from private subnets goes to NAT GW
    nat_gateway_id = aws_nat_gateway.main[0].id
  }
}

# Associate route tables with subnets (link the rule to the subnet)
resource "aws_route_table_association" "public" {
  count          = length(aws_subnet.public)
  subnet_id      = aws_subnet.public[count.index].id
  route_table_id = aws_route_table.public.id
}

resource "aws_route_table_association" "private" {
  count          = length(aws_subnet.private)
  subnet_id      = aws_subnet.private[count.index].id
  route_table_id = aws_route_table.private.id
}
STEP 4 — EKS CLUSTER
modules/eks/main.tf
# IAM Role for EKS Control Plane
resource "aws_iam_role" "eks_cluster" {
  name = "${var.project}-eks-cluster-role"

  assume_role_policy = jsonencode({
    Version   = "2012-10-17"
    Statement = [{
      Effect    = "Allow"
      Principal = { Service = "eks.amazonaws.com" }
      # eks.amazonaws.com = EKS service is allowed to ASSUME this role.
      # This is the trust policy — who can use this role.
      Action    = "sts:AssumeRole"
    }]
  })
}

# Attach required managed policies to the EKS cluster role
resource "aws_iam_role_policy_attachment" "eks_cluster_policy" {
  policy_arn = "arn:aws:iam::aws:policy/AmazonEKSClusterPolicy"
  role       = aws_iam_role.eks_cluster.name
}

# THE EKS CLUSTER RESOURCE
resource "aws_eks_cluster" "main" {
  name     = "${var.project}-${var.environment}"
  role_arn = aws_iam_role.eks_cluster.arn
  version  = var.k8s_version   # e.g., "1.28"

  vpc_config {
    subnet_ids              = concat(var.private_subnet_ids, var.public_subnet_ids)
    endpoint_private_access = true    # kubectl works from within VPC
    endpoint_public_access  = true    # kubectl works from internet (engineer's laptop)
    # For maximum security: endpoint_public_access = false
    # But then you need VPN or bastion to run kubectl
  }

  enabled_cluster_log_types = ["api", "audit", "authenticator"]
  # Logs go to CloudWatch. Cost: ~$0.50/GB. Essential for security auditing.

  depends_on = [aws_iam_role_policy_attachment.eks_cluster_policy]
}

# IAM Role for Worker Nodes
resource "aws_iam_role" "eks_nodes" {
  name = "${var.project}-eks-node-role"

  assume_role_policy = jsonencode({
    Version = "2012-10-17"
    Statement = [{
      Effect    = "Allow"
      Principal = { Service = "ec2.amazonaws.com" }
      Action    = "sts:AssumeRole"
    }]
  })
}

# 3 policies required for EKS worker nodes to function
locals {
  node_policies = [
    "arn:aws:iam::aws:policy/AmazonEKSWorkerNodePolicy",        # Connect to EKS
    "arn:aws:iam::aws:policy/AmazonEKS_CNI_Policy",             # Configure pod networking
    "arn:aws:iam::aws:policy/AmazonEC2ContainerRegistryReadOnly" # Pull from ECR
  ]
}

resource "aws_iam_role_policy_attachment" "node_policies" {
  # for_each with toset() — creates one resource per policy
  for_each   = toset(local.node_policies)
  policy_arn = each.value
  role       = aws_iam_role.eks_nodes.name
}

# MANAGED NODE GROUP — pool of EC2 worker nodes
resource "aws_eks_node_group" "main" {
  cluster_name    = aws_eks_cluster.main.name
  node_group_name = "main-workers"
  node_role_arn   = aws_iam_role.eks_nodes.arn
  subnet_ids      = var.private_subnet_ids   # Nodes in PRIVATE subnets
  instance_types  = ["t3.medium"]
  disk_size       = 50   # GB for node root volume

  scaling_config {
    desired_size = 2
    min_size     = 2
    max_size     = 10
  }

  update_config {
    # max_unavailable = how many nodes can be down during a cluster upgrade.
    # 1 = upgrade nodes one at a time. Slower but safer.
    max_unavailable = 1
  }

  labels = {
    role = "worker"
  }

  # lifecycle: prevent Terraform from replacing nodes when desired_size changes.
  # Cluster Autoscaler controls the actual count — don't let Terraform override!
  lifecycle {
    ignore_changes = [scaling_config[0].desired_size]
  }

  depends_on = [aws_iam_role_policy_attachment.node_policies]
}

# OIDC Provider — enables IRSA (IAM Roles for Service Accounts)
data "tls_certificate" "eks" {
  url = aws_eks_cluster.main.identity[0].oidc[0].issuer
}

resource "aws_iam_openid_connect_provider" "eks" {
  client_id_list  = ["sts.amazonaws.com"]
  thumbprint_list = [data.tls_certificate.eks.certificates[0].sha1_fingerprint]
  url             = aws_eks_cluster.main.identity[0].oidc[0].issuer
}

💬 INTERVIEW Q&A — Complete Answers for Project 5
What is IRSA (IAM Roles for Service Accounts) and why is it better than EC2 instance profiles?
EC2 Instance Profile problem: All pods on a node share the node's IAM role. If you give the node S3 access for one pod, ALL pods on that node get S3 access. This violates least-privilege.

IRSA solution: Each Kubernetes ServiceAccount gets its own IAM role. Only pods using that ServiceAccount can assume that role. Pod-level granularity.

How it works: EKS creates an OIDC identity provider. When a pod starts with an annotated ServiceAccount, it gets a signed JWT token projected into the pod. The application exchanges this token with AWS STS using AssumeRoleWithWebIdentity, getting temporary credentials scoped to that pod's role.

No credentials stored anywhere — tokens are temporary (default 86400 seconds), automatically rotated, and never appear in environment variables or config files.
What does EKS manage vs what do you still manage?
AWS manages (EKS control plane): API server, etcd, scheduler, controller-manager, control plane HA across 3 AZs, control plane version upgrades, etcd backups. You pay $0.10/hour for this.

You manage: Worker nodes (EC2 instance type, OS, patching — unless using managed node groups), node group scaling, Kubernetes add-ons (CoreDNS, kube-proxy, VPC CNI), application deployments, namespaces, RBAC, monitoring.

Managed Node Groups (what we used): AWS handles node OS patching, K8s version upgrades, draining nodes before replacement, launching new nodes. You only choose instance type and scaling bounds.
How does Terraform state work? What is state locking and why is it critical?
Terraform state is a JSON file that maps your Terraform resources to real AWS resources. Terraform uses it to know: "this aws_eks_cluster.main in my .tf file = arn:aws:eks:ap-south-1:123:cluster/fluxcart-prod in AWS".

Remote state (S3): Without remote state, state is a local file. Works for solo devs but breaks for teams — conflicting states if two people run apply.

State locking (DynamoDB): When terraform apply runs, it writes a lock record to DynamoDB. If another team member runs apply simultaneously, they see "Error acquiring state lock" and must wait. Without locking: both applies run, state gets corrupted, resources get duplicated or destroyed.

STEP 2 — BLUE/GREEN ROLLOUT
k8s/rollouts/bluegreen-rollout.yaml
# HOW BLUE/GREEN WORKS:
# Blue = current production version (all traffic)
# Green = new version (no traffic, fully running)
# Cutover = switch all traffic from Blue to Green instantly
# Rollback = switch back to Blue (still running!)
#
# The services:
# activeService: receives 100% of production traffic
# previewService: only Green pods (for testing before cutover)

apiVersion: argoproj.io/v1alpha1
kind: Rollout
metadata:
  name: backend-api
  namespace: fluxcart
spec:
  replicas: 3
  selector:
    matchLabels:
      app: backend-api
  template:
    metadata:
      labels:
        app: backend-api
    spec:
      containers:
        - name: backend
          image: fluxcart/backend:v1.0.0
          ports:
            - containerPort: 4000

  strategy:
    blueGreen:
      # activeService = the K8s Service that routes to ACTIVE (Blue) pods.
      # This is what users hit. It gets updated on promotion.
      activeService: backend-api-active
      # previewService = K8s Service that routes to NEW (Green) pods.
      # QA team tests via this service BEFORE promotion to production.
      previewService: backend-api-preview

      # autoPromotionEnabled: false = MANUAL approval required to promote.
      # true = automatically promote Green to Blue after autoPromotionSeconds.
      # Use false in production — never auto-promote without human review!
      autoPromotionEnabled: false

      # scaleDownDelaySeconds: after promotion, keep Blue pods running for 60s.
      # Why? If Green has issues immediately after promotion, instant rollback!
      # Without this: Blue is gone, rollback takes time to spin up new pods.
      scaleDownDelaySeconds: 60

---
# Services for Blue/Green
apiVersion: v1
kind: Service
metadata:
  name: backend-api-active    # Production traffic
  namespace: fluxcart
spec:
  selector:
    app: backend-api
  ports:
    - port: 4000
      targetPort: 4000

---
apiVersion: v1
kind: Service
metadata:
  name: backend-api-preview    # Preview traffic (QA only)
  namespace: fluxcart
spec:
  selector:
    app: backend-api
  ports:
    - port: 4000
      targetPort: 4000

STEP 4 — CANARY WITH PROMETHEUS ANALYSIS
k8s/rollouts/canary-rollout.yaml + analysis-template.yaml
# AnalysisTemplate: defines WHAT metrics to check and WHEN to rollback
apiVersion: argoproj.io/v1alpha1
kind: AnalysisTemplate
metadata:
  name: success-rate-analysis
  namespace: fluxcart
spec:
  metrics:
    - name: success-rate
      interval: 1m        # Query Prometheus every minute
      count: 5            # Must pass 5 times before step completes
      # successCondition: CONTINUE if result >= 0.95 (95% success rate)
      successCondition: result[0] >= 0.95
      # failureCondition: ROLLBACK immediately if result < 0.90
      failureCondition: result[0] < 0.90
      provider:
        prometheus:
          address: http://prometheus-operated.monitoring:9090
          query: |
            sum(rate(http_requests_total{
              app="backend-api",
              rollout_type="canary",
              status!~"5.."
            }[5m]))
            /
            sum(rate(http_requests_total{
              app="backend-api",
              rollout_type="canary"
            }[5m]))

---
# Canary Rollout
apiVersion: argoproj.io/v1alpha1
kind: Rollout
metadata:
  name: backend-api-canary
  namespace: fluxcart
spec:
  replicas: 5    # 5 total pods (1 canary = 20%, 4 stable = 80%)
  selector:
    matchLabels:
      app: backend-api
  template:
    metadata:
      labels:
        app: backend-api
    spec:
      containers:
        - name: backend

💬 INTERVIEW Q&A — Complete Answers for Project 6
Explain Blue/Green vs Canary vs Rolling Update. When would you use each?
Rolling Update (K8s default): Replaces pods gradually. Old and new versions run simultaneously during update. Simplest, no extra infrastructure. Use for: most deployments, stateless apps with backward-compatible changes.

Blue/Green: Full duplicate environment. Instant cutover. Both versions run simultaneously but only one receives traffic. Use for: zero-downtime critical releases, when you need instant rollback, when old/new versions can't coexist (breaking API changes).

Canary: New version gets small % of traffic, gradually increased. Real users test new version. Use for: high-risk changes, A/B testing, when you want data before full rollout.
What are the infrastructure costs of Blue/Green deployment? What's the tradeoff?
Cost: Blue/Green requires running 2x the normal infrastructure during deployment — double the pods, double the CPU/memory. For a $500/month cluster, a 30-minute deploy costs about $10 extra.

Tradeoffs:
✅ Instant rollback (Blue still running)
✅ Zero downtime during cutover
✅ QA can test Green in production before promotion
❌ 2x infrastructure cost during deployment
❌ Database schema changes must be backward-compatible (both versions hit same DB)

Database problem: If v2 adds a new required column, v1 will fail writing records. Solution: expand/contract migrations — add column as nullable (both versions work), then after full migration to v2, add NOT NULL constraint.

💬 INTERVIEW Q&A — Complete Answers for Project 7
What does "idempotent" mean in Ansible? Why is it important?
Idempotent means running the operation once or ten times produces the same result. Example: state: present for a package — if it's installed, nothing changes (ok), if not installed, it installs (changed).

Why critical: You can safely re-run playbooks without worrying about breaking things. Example uses: (1) recover from partial failures by re-running the whole playbook, (2) run on a schedule (like every hour) to enforce configuration, (3) add new servers to an existing fleet without writing server-specific logic.

Ansible's built-in modules ARE idempotent. But if you use shell: or command:, you must add creates:, removes:, or when: conditions to make them idempotent yourself.
When would you use Ansible vs Terraform? Are they competing tools?
They're complementary, not competing:

Terraform = Infrastructure provisioning. Creates/destroys AWS resources (EC2, VPC, RDS, S3). Declarative state management. Think: "provision this EC2 instance."

Ansible = Configuration management. Configures what's running inside those resources (install software, deploy code, manage files, create users). Think: "after the EC2 exists, install Nginx on it."

Typical workflow: Terraform creates VPC + EC2 instances → Ansible configures the EC2 instances with software → your CI/CD deploys application code.

Overlap: Terraform has provisioner blocks (can run scripts). Ansible has cloud modules (can create EC2). But each is better at its primary purpose.
What is the difference between a Task, Handler, Role, and Playbook?
Task: Single unit of work. Calls an Ansible module. Example: "Install nginx using the package module."

Handler: A special task triggered by notify. Runs ONCE at end of play, even if notified multiple times. Used for: service restarts, config reloads. Avoids restarting nginx 5 times if 5 tasks change config files.

Role: Reusable unit of organization. A directory with predefined structure (tasks/, handlers/, templates/, files/, vars/). A role for "nginx" can be used across 10 playbooks without copy-pasting.

Playbook: The top-level YAML file. Defines: which hosts to target (hosts:), which roles to apply (roles:), and in what order. Think of it as the "orchestrator" that combines roles.

policies/kubernetes.rego — Security Policies
# Rego is OPA's policy language.
# deny[msg]: if the condition is TRUE, the policy FAILS with msg.
# warn[msg]: if the condition is TRUE, show warning (not fail).
# Conftest runs all deny/warn rules and reports results.

package main

# ─ Policy 1: No :latest image tags ──────────────────────────────
# :latest = non-deterministic (what does "latest" mean next week?)
# Always pin to specific version: image:v1.2.3
deny[msg] {
  input.kind == "Deployment"
  container := input.spec.template.spec.containers[_]  # [_] = any container
  # endswith: check if string ends with ":latest" OR has no tag (defaults to latest)
  endswith(container.image, ":latest")
  msg := sprintf("DENY: Container '%v' uses :latest tag. Pin to specific version!", [container.name])
}

# ─ Policy 2: Resource limits required ────────────────────────────
# Without limits: one runaway container can OOM the entire node,
# killing ALL other pods. Always set CPU and memory limits.
deny[msg] {
  input.kind == "Deployment"
  container := input.spec.template.spec.containers[_]
  not container.resources.limits.memory    # "not" in Rego = condition is absent
  msg := sprintf("DENY: Container '%v' has no memory limit. Risk: OOMKill entire node!", [container.name])
}

deny[msg] {
  input.kind == "Deployment"
  container := input.spec.template.spec.containers[_]
  not container.resources.limits.cpu
  msg := sprintf("DENY: Container '%v' has no CPU limit. Risk: CPU starvation!", [container.name])
}

# ─ Policy 3: No root containers ──────────────────────────────────
deny[msg] {
  input.kind == "Deployment"
  container := input.spec.template.spec.containers[_]
  container.securityContext.runAsUser == 0
  msg := sprintf("DENY: Container '%v' runs as root (UID 0). Security violation!", [container.name])
}

# ─ Policy 4: Readiness probe required ────────────────────────────
deny[msg] {
  input.kind == "Deployment"
  container := input.spec.template.spec.containers[_]
  not container.readinessProbe
  msg := sprintf("DENY: Container '%v' has no readinessProbe. App will receive traffic before ready!", [container.name])
}

# ─ Policy 5: Namespace must be specified ─────────────────────────
deny[msg] {
  input.kind == "Deployment"
  not input.metadata.namespace
  msg = "DENY: Deployment must specify a namespace. Never deploy to 'default'!"
}

# ─ WARNING: Single replica (not a hard error, but risky) ─────────
warn[msg] {
  input.kind == "Deployment"
  input.spec.replicas == 1
  msg = "WARN: Only 1 replica. Node failure = downtime. Consider replicas >= 2"
}

💬 INTERVIEW Q&A — Complete Answers for Project 8
What is "shift left" security? How does your DevSecOps pipeline implement it?
"Shift left" means moving security checks earlier in the development lifecycle. "Left" = earlier in the pipeline (code → commit → PR → build → deploy → production).

Traditional (shift right): Security team scans production every quarter. Vulnerabilities found in production are expensive: code is deployed, customers may be impacted, fixing requires emergency patches.

Shift left implementation in this pipeline:
— Secret scan: on every commit (pre-push hook / CI)
— Dependency CVEs: on every PR (before code review)
— Image scan: before image is pushed to ECR
— Policy checks: before manifests reach K8s cluster

Result: Developer finds and fixes a vulnerability in their own PR, in minutes, before it ever reaches production.
What is the difference between SAST, DAST, and SCA? Give tools for each.
SAST (Static Application Security Testing): Analyzes source code without running it. Finds: SQL injection patterns, hardcoded credentials, insecure functions. Tools: Semgrep, SonarQube, CodeQL. Runs early — no running app needed.

DAST (Dynamic Application Security Testing): Tests the running application by sending crafted requests. Finds: XSS, CSRF, authentication flaws, open redirects. Tools: OWASP ZAP, Burp Suite. Requires deployed app.

SCA (Software Composition Analysis): Analyzes third-party dependencies (npm, pip, maven) for known CVEs. Tools: Trivy, Snyk, OWASP Dependency-Check. Most critical for modern apps — 80% of code is third-party.
What is an SBOM? Why do enterprises and governments require it?
SBOM (Software Bill of Materials) is a complete inventory of every component in your software: open-source libraries, their versions, their licenses, and their transitive dependencies.

Why governments require it: US Executive Order 14028 (2021) mandates SBOMs for software sold to the US government after the SolarWinds supply chain attack. Attackers compromised SolarWinds update mechanism — SBOMs make it possible to quickly identify ALL software containing a compromised component.

Practical uses:
— When Log4Shell was announced: query your SBOM database to instantly find all apps containing log4j
— License compliance: ensure no GPL code in proprietary products
— Vendor audits: enterprise customers require proof of software components

Formats: SPDX (Linux Foundation standard) and CycloneDX (OWASP standard) are the two main formats.
A Critical CVE is found in a production dependency. Can you still deploy? What's the process?
Not a binary yes/no — it depends on exploitability:

1. Is there a fix available? If yes: upgrade immediately, re-run pipeline, deploy.
2. Is the vulnerable code path reachable? A CVE in an image-processing library doesn't matter if your app never processes images. Document this as a "risk acceptance."
3. What is the CVSS vector? Network-accessible with no auth required = highest priority. Local access only = lower priority.
4. If no fix available: Use trivy --ignore-unfixed to exclude unfixable CVEs. Open a tracking issue. Add to a CVE exception file with justification.

The process: Security team + product owner review → risk acceptance decision → documented exception → periodic review (CVE might get a fix later).
