The Full Pipeline Flow

🔄 End-to-End: One Code Push to Production in 23 Minutes

.github/workflows/ci-pipeline.yml — The Master OrchestratorYAML — Complete Pipeline
name: CloudMart India — Full CI/CD Pipeline
on:
  push:
    branches: [main, develop]
    paths: ['services/**', 'db/**']
  pull_request:
    branches: [main]

env:
  ECR_REGISTRY:   ${{ secrets.AWS_ACCOUNT_ID }}.dkr.ecr.ap-south-1.amazonaws.com
  EKS_CLUSTER:    cloudmart-prod-cluster
  AWS_REGION:     ap-south-1

jobs:

    # ── STAGE 1: QUALITY (parallel — fastest feedback) ────────────── ~3 min
  unit-test:
    name: Unit Tests + Coverage
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '20', cache: 'npm' }
      - run: npm ci && npm run test:coverage

  sonar-scan:
    name: SonarQube Quality Gate
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with: { fetch-depth: 0 }
      - uses: SonarSource/sonarqube-scan-action@master
        env: { SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }}, SONAR_HOST_URL: ${{ secrets.SONAR_HOST_URL }} }
      - uses: SonarSource/sonarqube-quality-gate-action@master
        env: { SONAR_TOKEN: ${{ secrets.SONAR_TOKEN }} }

    # ── STAGE 2: SECURITY SCANS (parallel with stage 1) ──────────── ~4 min
  dependency-scan:
    name: Snyk Dependency CVE Scan
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: snyk/actions/node@master
        env: { SNYK_TOKEN: ${{ secrets.SNYK_TOKEN }} }
        with: { args: --severity-threshold=high }

    # ── STAGE 3: BUILD + IMAGE SCAN ───────────────────────────────── ~6 min
  build-scan-push:
    name: Build → Trivy Scan → Push to ECR
    runs-on: ubuntu-latest
    needs: [unit-test, sonar-scan, dependency-scan]
    outputs:
      image-tag: ${{ github.sha }}
    steps:
      - uses: actions/checkout@v4
      - uses: docker/setup-buildx-action@v3
      - uses: aws-actions/configure-aws-credentials@v4
        with: { role-to-assume: ${{ secrets.AWS_ROLE_ARN }}, aws-region: ${{ env.AWS_REGION }} }
      - run: docker build -t cloudmart-app:scan .
      - uses: aquasecurity/trivy-action@master
        with: { image-ref: cloudmart-app:scan, severity: CRITICAL,HIGH, exit-code: '1', ignore-unfixed: true }
      - uses: aws-actions/amazon-ecr-login@v2
      - uses: docker/build-push-action@v5
        with:
          push: true
          tags: ${{ env.ECR_REGISTRY }}/cloudmart-app:${{ github.sha }},${{ env.ECR_REGISTRY }}/cloudmart-app:latest
          cache-from: type=gha
          cache-to: type=gha,mode=max

    # ── STAGE 4: DB MIGRATION + STAGING DEPLOY ────────────────────── ~8 min
  db-migrate-staging:
    name: Database Migration — Staging
    runs-on: ubuntu-latest
    needs: build-scan-push
    environment: staging
    steps:
      - uses: actions/checkout@v4
      - uses: docker://flyway/flyway:10-alpine
        with:
          args: -url=jdbc:postgresql://${{ secrets.STAGING_DB_HOST }}:5432/cloudmart -user=${{ secrets.STAGING_DB_USER }} -password=${{ secrets.STAGING_DB_PASSWORD }} -locations=filesystem:db/migrations migrate

  deploy-staging:
    name: Deploy to Staging EKS
    runs-on: ubuntu-latest
    needs: db-migrate-staging
    environment: staging
    steps:
      - uses: actions/checkout@v4
      - uses: aws-actions/configure-aws-credentials@v4
        with: { role-to-assume: ${{ secrets.STAGING_DEPLOY_ROLE }}, aws-region: ${{ env.AWS_REGION }} }
      - run: aws eks update-kubeconfig --name cloudmart-staging-cluster --region ${{ env.AWS_REGION }}
      - run: sed -i "s|IMAGE_TAG|${{ github.sha }}|g" k8s/overlays/staging/kustomization.yaml
      - run: kubectl apply -k k8s/overlays/staging/
      - run: kubectl rollout status deployment/cloudmart-app -n cloudmart-staging --timeout=5m

    # ── STAGE 5: INTEGRATION + E2E TESTS ON STAGING ──────────────── ~5 min
  integration-tests:
    name: Integration + E2E Tests on Staging
    runs-on: ubuntu-latest
    needs: deploy-staging
    steps:
      - run: npm run test:integration && npm run test:e2e
        env: { TEST_URL: ${{ secrets.STAGING_URL }} }

    # ── STAGE 6: PRODUCTION (gated, main branch only) ─────────────── ~5 min
  deploy-production:
    name: Deploy to Production EKS
    runs-on: ubuntu-latest
    needs: integration-tests
    if: github.ref == 'refs/heads/main'
    environment: production
    steps:
      - uses: actions/checkout@v4
      - uses: aws-actions/configure-aws-credentials@v4
        with: { role-to-assume: ${{ secrets.PROD_DEPLOY_ROLE }}, aws-region: ${{ env.AWS_REGION }} }
      - run: aws eks update-kubeconfig --name ${{ env.EKS_CLUSTER }} --region ${{ env.AWS_REGION }}
      - run: sed -i "s|IMAGE_TAG|${{ github.sha }}|g" k8s/overlays/production/kustomization.yaml
      - run: kubectl apply -k k8s/overlays/production/
      - run: kubectl rollout status deployment/cloudmart-app -n cloudmart-prod --timeout=5m

    # ── STAGE 7: POST-DEPLOY VALIDATION + NOTIFY ─────────────────── ~2 min
  post-deploy:
    name: Post-Deploy Monitoring Check + Notify
    runs-on: ubuntu-latest
    needs: deploy-production
    if: always()
    steps:
      - name: Wait and check error rate
        if: needs.deploy-production.result == 'success'
        run: |
          sleep 300
          ERROR_RATE=$(curl -s "${{ secrets.PROMETHEUS_URL }}/api/v1/query?query=sum(rate(http_requests_total{status=~'5..',app='cloudmart'}[5m]))/sum(rate(http_requests_total{app='cloudmart'}[5m]))*100" | jq -r '.data.result[0].value[1]')
          if (( $(echo "$ERROR_RATE > 1.0" | bc -l) )); then
            kubectl rollout undo deployment/cloudmart-app -n cloudmart-prod
            exit 1
          fi
      - name: Slack notification
        uses: slackapi/slack-github-action@v1.27.0
        with:
          channel-id: ${{ needs.deploy-production.result == 'success' && secrets.SLACK_DEPLOY_CHANNEL || secrets.SLACK_ALERT_CHANNEL }}
          payload: '{"text":"${{ needs.deploy-production.result == ''success'' && ''✅ Production deployed'' || ''❌ Deployment FAILED'' }}: ${{ github.sha }} by ${{ github.actor }}"}'
        env: { SLACK_BOT_TOKEN: ${{ secrets.SLACK_BOT_TOKEN }} }

The Golden Rule

📌 The One Rule for Every HR Answer

Talk about what you did, what you built, what you achieved — not what you "know about" or "have studied". HR is checking three things: can you communicate clearly, are you stable and positive, and will you join quickly. Every answer should show these three things, calmly and confidently.

Introductions

🗣️ "Tell Me About Yourself" — The Most Important Answer

Q: "Tell me about yourself" / "Introduce yourself"

❌ Wrong — Never Say This

"Myself Puneeth Kumar. I am from Bengaluru. I completed my engineering in 2020. I have total 4 years of experience. I know Docker, Kubernetes, Jenkins, GitHub Actions, Terraform, AWS, and many other tools..."

✅ Say This Instead — Confident, Story-Based, 90 Seconds

"Good morning. My name is Puneeth Kumar. I'm a DevOps and Cloud Engineer with four years of experience, currently based in Bengaluru.

I started my career in support engineering — working with SQL Server, SSIS, ETL pipelines, and Windows Server administration. That gave me a strong foundation in how production systems actually behave under real workloads.

Over the last three years, I've been working in DevOps — building and maintaining CI/CD pipelines using Jenkins and GitHub Actions, managing infrastructure on AWS using Terraform and Ansible, containerising applications with Docker, and working with Kubernetes for deployments and orchestration. I've also worked with monitoring tools like Prometheus and Grafana.

I've worked across both telecom and e-commerce domains, so I'm comfortable with fast deployment cycles and on-call responsibility. I'm now looking for a role where I can go deeper into cloud-native DevOps — specifically around Kubernetes, GitOps, and larger-scale AWS architectures."

💬 Structure: Name → Years + Location → Past experience → Current skills → What you want next. Never more than 90 seconds. Never list tools like a grocery list.

Why Leaving

🔄 "Why Are You Looking for a Change?"

Q: Why are you looking for a job change?

❌ Wrong — Never Say This

"My current company has no growth. Manager is not good. Salary is less. Work-life balance is bad. Too much politics."

✅ Say This — Positive and Forward-Looking

"I've had a good experience at my current company and learned a lot — especially around CI/CD pipelines and AWS infrastructure. But after four years, I feel I've gone as far as I can go in my current role in terms of the type and scale of work available to me. I'm at a point where I want to work with larger Kubernetes clusters, more complex multi-region architectures, and teams that are operating at a higher scale of deployments per day. I'm looking for an environment where I can continue growing technically and take on more ownership of the DevOps platform as a whole."

💬 Always: positive about the past, clear about what you want next. Never speak negatively about your current company — it makes interviewers wonder what you'll say about them someday.

Q: Why did you complete your notice period already / what have you been doing since leaving?

✅ Say This

"Yes, I've completed my notice period and I'm actively interviewing. During this time I've been focusing on deepening my skills — I've been building a complete CI/CD project covering GitHub Actions, Kubernetes deployments, ArgoCD GitOps, and Terraform, which I can show you on GitHub. I've also been specifically targeting roles in companies like yours where the scale of infrastructure work matches where I want to go next."

💬 The capstone project from Chapter 25 is your honest, concrete answer to "what have you been doing" — make sure the repo is on GitHub before interviews.

📋 More HR Questions — Answered Naturally

"My strongest area is building CI/CD pipelines from scratch — I enjoy taking a team from manual, error-prone deployments to a fully automated, monitored pipeline. I also think my background in both support engineering and DevOps gives me a useful perspective — I understand how production issues feel from both the user side and the infrastructure side, which helps me design systems with the right alerting and recovery mechanisms built in."

"I've noticed that when I'm deep in solving a technical problem, I sometimes spend more time than needed trying to find the most elegant solution before asking for help or moving on. "In five years, I'd like to be working at the Senior or Lead DevOps Engineer level — owning the platform engineering strategy for an organisation, not just maintaining pipelines but designing them. Q: Why do you want to join our company specifically?

✅ Say This — Do research before every interview

"I spent some time researching the company before this call. I saw that [company] recently [launched X product / expanded to Y cities / raised Series C funding / moved to a microservices architecture based on your engineering blog]. That tells me the infrastructure and deployment challenges here are at the scale where my skills — Kubernetes, CI/CD automation, AWS — would be genuinely useful and stretched, rather than just maintaining what already works. I'm looking for that kind of challenge."

💬 Always research the company for 20 minutes before the call. Check LinkedIn, their engineering blog, and recent news. One specific detail from your research — any specific detail — makes you stand out from 90% of candidates.

✅ Always Say Yes — These Are Good Questions

"Yes, I have a few. What does the current CI/CD maturity look like on the team — are most deployments already automated, or is there a migration in progress? 💬 Always have 2-3 questions ready. Never say "I don't have any questions" — it signals low interest. Questions about the team structure, technical stack depth, and growth opportunities are always well-received.

Market Rates

💰 DevOps/Cloud Engineer Salary Ranges — Bengaluru 2025

These are approximate ranges based on market data for Bengaluru. Actual offers vary by company type (startup vs MNC), domain (fintech vs e-commerce vs IT services), and your specific skills (Kubernetes + AWS commands more than Jenkins alone).

Role / ExperienceIT Services / Product ServiceProduct / StartupMNC / Fintech
Junior DevOps (1–2 YOE)₹6–10 LPA₹8–14 LPA₹10–16 LPA
DevOps Engineer (3–5 YOE)₹12–18 LPA₹16–28 LPA₹22–38 LPA

Note: LPA = Lakhs Per Annum (CTC). These are CTC figures, not take-home. Take-home is typically 65–75% of CTC after PF, professional tax, and income tax deductions. ESOP/stocks not included in these figures — at funded startups, stocks can add significant value above these ranges.

Understanding Your CTC Components

Fixed Components (You Always Get)

  • Basic Salary — 40–50% of CTC, taxable
  • HRA — House Rent Allowance, partially tax-exempt if paying rent
  • Special Allowance — makes up the rest of the monthly salary
  • LTA — Leave Travel Allowance, 1–2x per year

Variable / Conditional Components

  • Variable Pay / Bonus — typically 10–20% of CTC, performance-linked. You may NOT get 100% of this.
  • PF (Employer's contribution) — 12% of Basic, included in CTC but goes to your PF account, not salary
  • Gratuity — only payable after 5 continuous years at the same company
  • Joining Bonus — one-time, often with clawback if you leave within 1 year

⚠️ The Variable Pay Trap — Read This Carefully

If a company offers ₹20 LPA with ₹4 LPA variable pay, your guaranteed fixed is ₹16 LPA. The ₹4 LPA variable depends on your performance rating and the company's performance. In a bad quarter or year, you may receive 0–50% of the variable. Always calculate and compare offers on their fixed component alone, then treat variable as a bonus. Never count on variable to pay your rent.

Salary Questions

🤝 How to Handle Salary Questions

Q: What is your current CTC?

✅ Say This — Be Honest and Precise

"My current CTC is ₹X LPA, which includes ₹Y LPA fixed and ₹Z LPA variable. My take-home per month is approximately ₹[amount]."

💬 Always state fixed and variable separately. Interviewers know this distinction. Saying only the total CTC when it includes high variable can mislead them and hurt negotiations later when they verify documents.

Q: What are your salary expectations?

❌ Wrong — Avoid These

"Whatever you offer, I am okay." (gives away all leverage) / "I want 50% hike minimum." (without basis) / Saying a number before you know the role scope fully.

✅ Say This — Anchor High, Stay Flexible

"Based on my research for this role and location, I'm looking at something in the range of ₹[your target] to ₹[10% above target]. That said, I'm open to discussion — I'm more interested in the right role and team fit than hitting an exact number. Could you share what the band for this position looks like on your end?"

💬 Always try to make them say a number first. Once they reveal the band, you know if your target is realistic or you can push higher. Whoever says a number first is usually at a disadvantage.

Q: The offer is ₹X — is that okay for you? / We can only go up to ₹X.

✅ Counter-Offer Script — Calm and Specific

"Thank you for the offer — I appreciate it and I'm genuinely excited about this role. I was expecting something closer to ₹[your target]. Given my experience with [mention 2 specific relevant skills — e.g., 'Kubernetes and AWS EKS at scale' or 'Jenkins shared library architecture'], I believe that level is fair for the value I'll bring. Is there any flexibility to move closer to that number? Even ₹[a middle point] would help me make a confident decision quickly."

💬 Three things make a counter-offer work: (1) You express genuine interest — you're not bluffing. (2) You cite specific skills, not just "I have experience". (3) You give them a concrete number to agree to. "Any flexibility" without a number gives them nothing to work with.

Notice Period

📅 Notice Period Negotiation

Notice Period Buyout — How It Works

Some companies will pay your notice period salary on your behalf — called "buying out" your notice. This lets you join faster. Typical range: 15–45 days' salary. Not all companies offer this. Ask during the offer negotiation: "Is there a provision for notice period buyout if I'm able to negotiate an early release?"

Getting Released Early — What You Can Do

Talk to your manager honestly: "I've received an offer and the new company needs me earlier. I've documented my work and am happy to do a thorough knowledge transfer. Is an early release possible?" A good manager often says yes, especially if work is well-documented. Start this conversation early — not on your last week.

Q: Your notice period is 90 days — we need someone in 30 days. Can you join sooner?

✅ Say This

"My official notice period is 90 days, but I'm going to negotiate with my current manager for an early release. I'm confident I can complete a proper knowledge transfer quickly given how well-documented my work is. I'll try my best to reduce it to 30–45 days. If that's not possible, would you be open to a buyout arrangement, or is there any flexibility on the joining date on your end? I want to make this work."

Evaluating an Offer

📄 What to Check Before Signing an Offer Letter

Offer Letter Checklist — Read Every Word

  1. Is the fixed CTC clearly split — Basic, HRA, Special Allowance, LTA separately listed?
  2. What is the exact variable pay percentage and what are the performance criteria for 100% payout?
  3. Is there a joining bonus? What is the clawback period — what happens if you leave in 6 months?
  4. What is the probation period (usually 3–6 months) and is salary different during probation?
  5. What are the ESOP terms — vesting schedule, cliff, strike price? (For startups)
  6. What is the official notice period for you (and for them if they let you go)?
  7. What are the grounds for termination and what notice do they have to give you?
  8. Is there a non-compete clause or non-solicitation clause? For how long and how broadly defined?
  9. What is the work location and WFH policy stated in writing (not just verbally)?
  10. Are the role title and grade/band clearly stated — not just an informal title?

Pre-Interview Ritual

📋 30 Minutes Before Every Interview

  1. Open the company's website and LinkedIn — read one recent engineering blog post or news item
  2. Say your 90-second introduction out loud, alone, twice
  3. Review your 3 questions to ask the interviewer
  4. Open your GitHub portfolio repo — know the URL and be ready to share screen
  5. Drink water — your voice sounds better when hydrated
  6. Find a quiet place with good mobile signal or stable Wi-Fi — test audio quality with a friend if it's a video call
  7. Remind yourself: you have 4 years of real experience. You are not a student memorising answers. You are a professional sharing what you know.
Career Roadmap

🗺️ DevOps Career Path — From Here to Principal Engineer

Now — 4 YOE
DevOps / Cloud Engineer

You are here. Building pipelines, managing Kubernetes, writing Terraform, securing deployments. Interview target: ₹16–28 LPA at product companies, ₹22–38 LPA at MNCs/fintech in Bengaluru.

6–7 YOE · 18–24 months away
Senior DevOps / Cloud Engineer

Leading pipeline architecture decisions. Mentoring junior engineers. Owning the full CI/CD platform. Driving DORA metric improvements. Add: advanced Kubernetes (service mesh, KEDA, Karpenter), multi-region AWS architecture, FinOps. Target: ₹28–45 LPA.

8–10 YOE · 3–4 years away
Staff / Lead DevOps Engineer / Platform Engineer

Defining engineering standards across teams. Building Internal Developer Platforms (IDPs). Driving the 0→1 on DevOps culture at your organisation. Add: Backstage IDP, platform product thinking, cross-team technical strategy, SRE practices. Target: ₹45–70 LPA.

12+ YOE · 6–8 years away
Principal Engineer / DevOps Architect

Setting the multi-year technical vision for infrastructure and delivery. Evaluating and adopting new technologies at the company level. Public presence: conference talks, blog posts, open source contributions. Target: ₹80–120 LPA and beyond.

Parallel track — any time
Engineering Manager / Director of Engineering

If you enjoy growing people as much as systems — this is the path. Requires deliberate move from IC to people management. Best approached after Senior Engineer level when you have enough technical credibility to lead technical teams.
