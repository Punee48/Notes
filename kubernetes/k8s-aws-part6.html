    pod-security.kubernetes.io/warn=restricted
done

# Apply restricted PSS to payments namespace (highest security)
kubectl create namespace shopnow-payments

YAMLorders-deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: orders-service
  namespace: shopnow
  labels:
    app: orders-service
    version: "2.4.1"
    team: backend
spec:
  replicas: 3
  selector:
    matchLabels:
      app: orders-service
  strategy:
    type: RollingUpdate
    rollingUpdate:
      maxSurge: 1
      maxUnavailable: 0          # ZERO downtime — always 3 pods running
  template:
    metadata:
      labels:
        app: orders-service
        version: "2.4.1"
      annotations:
        prometheus.io/scrape: "true"   # Prometheus discovers this pod for metrics
        prometheus.io/port: "9090"
        prometheus.io/path: "/metrics"
    spec:
      serviceAccountName: orders-sa    # IRSA role: SQS read, Secrets Manager read
      
      affinity:
        podAntiAffinity:               # SPREAD pods across AZs — never 2 on same node
          requiredDuringSchedulingIgnoredDuringExecution:
          - labelSelector:
              matchLabels:
                app: orders-service
            topologyKey: kubernetes.io/hostname  # Different nodes
          preferredDuringSchedulingIgnoredDuringExecution:
          - weight: 100
            podAffinityTerm:
              labelSelector:
                matchLabels:
                  app: orders-service
              topologyKey: topology.kubernetes.io/zone  # Prefer different AZs

      tolerations:                     # Allow scheduling on cpu-optimized nodes
      - key: dedicated
        value: cpu-workload
        operator: Equal
        effect: NoSchedule

      topologySpreadConstraints:       # Ensure even spread across AZs
      - maxSkew: 1
        topologyKey: topology.kubernetes.io/zone
        whenUnsatisfiable: DoNotSchedule
        labelSelector:
          matchLabels:
            app: orders-service

      initContainers:
      - name: wait-for-db              # Don't start until DB is ready
        image: busybox:1.28
        command: ['sh','-c','until nc -z postgresql.shopnow-data.svc.cluster.local 5432; do echo waiting for db; sleep 2; done']

      containers:
      - name: orders-service
        image: 123456789.dkr.ecr.ap-south-1.amazonaws.com/shopnow/orders:2.4.1
        ports:
        - name: http
          containerPort: 3000
        - name: metrics
          containerPort: 9090

        env:
        - name: NODE_ENV
          value: production
        - name: PORT
          value: "3000"
        - name: DB_HOST
          value: postgresql.shopnow-data.svc.cluster.local
        - name: DB_NAME
          value: shopnow_orders
        - name: REDIS_HOST
          value: redis.shopnow-data.svc.cluster.local
        - name: SQS_QUEUE_URL
          valueFrom:
            configMapKeyRef:
              name: orders-config
              key: SQS_QUEUE_URL
        - name: DB_PASSWORD
          valueFrom:
            secretKeyRef:
              name: db-secret
              key: password

        resources:
          requests:
            cpu: "250m"
            memory: "256Mi"
          limits:
            cpu: "1000m"
            memory: "512Mi"

        livenessProbe:
          httpGet:
            path: /health/live
            port: 3000
          initialDelaySeconds: 20
          periodSeconds: 10
          failureThreshold: 3          # 3 consecutive failures → restart

        readinessProbe:
          httpGet:
            path: /health/ready
            port: 3000
          initialDelaySeconds: 10
          periodSeconds: 5
          successThreshold: 1          # 1 success = mark ready again

        startupProbe:                  # Give slow-starting apps extra time
          httpGet:
            path: /health/live
            port: 3000
          failureThreshold: 30         # 30 × 2s = 60s max startup time
          periodSeconds: 2

        lifecycle:
          preStop:                     # Graceful shutdown: wait 15s before SIGTERM
            exec:
              command: ["/bin/sh","-c","sleep 15"]

        securityContext:
          runAsNonRoot: true
          runAsUser: 1001
          allowPrivilegeEscalation: false
          readOnlyRootFilesystem: true
          capabilities:
            drop: ["ALL"]

        volumeMounts:
        - mountPath: /tmp
          name: tmp

      terminationGracePeriodSeconds: 60   # Give app 60s to drain connections

      volumes:
      - name: tmp
        emptyDir: {}
---
# PodDisruptionBudget — guarantee availability during node drains
apiVersion: policy/v1
kind: PodDisruptionBudget
metadata:
  name: orders-pdb
  namespace: shopnow
spec:
  minAvailable: 2                      # During drain: at least 2 pods must stay running
  selector:
    matchLabels:
      app: orders-service

Word-by-Word: Production-Critical Fields

podAntiAffinity: requiredDuring...
Why this is essential: Without anti-affinity, the scheduler could place all 3 orders pods on the same node. If that node fails → zero availability. requiredDuringScheduling is a hard rule — the scheduler WILL NOT place two orders pods on the same node, even if it means the pod stays Pending.
topologySpreadConstraints
AZ spread: Ensures pods are spread evenly across AZs (ap-south-1a, 1b, 1c). maxSkew: 1 means the difference between the most and least loaded AZ cannot exceed 1 pod. Together with podAntiAffinity (node-level), this gives you true multi-AZ high availability.
startupProbe
What it solves: Java/JVM and Node apps can take 30–60 seconds to start. Without startupProbe, livenessProbe fires before the app is ready and kills the container — infinite restart loop. startupProbe gives the app up to 60 seconds to start. Once it passes, liveness and readiness probes take over.
lifecycle.preStop
Graceful shutdown: When K8s decides to stop a pod (rolling update, node drain), it removes the pod from Service endpoints AND sends SIGTERM simultaneously. But clients may still send requests for a few seconds. The preStop sleep(15) delays SIGTERM by 15s, giving time for in-flight requests to complete and load balancers to stop routing to this pod.
PodDisruptionBudget
Why it matters: When you run kubectl drain (node maintenance) or Cluster Autoscaler scales down, Kubernetes evicts pods. Without a PDB, it could evict ALL pods simultaneously → outage. PDB with minAvailable: 2 ensures at least 2 orders pods run during disruption — only 1 can be evicted at a time.

Phase 3

GitHub Actions → ECR → ArgoCD

YAML.github/workflows/deploy.yml
name: Build and Deploy Orders Service

on:
  push:
    branches: [main]
    paths: ['services/orders/**']  # Only trigger on changes to orders service

env:
  AWS_REGION: ap-south-1
  ECR_REGISTRY: 123456789.dkr.ecr.ap-south-1.amazonaws.com
  IMAGE_NAME: shopnow/orders

jobs:
  build-and-push:
    runs-on: ubuntu-latest
    permissions:
      id-token: write       # Required for OIDC auth to AWS (no long-lived keys!)
      contents: read

    steps:
    - name: Checkout code
      uses: actions/checkout@v4

    - name: Configure AWS credentials (OIDC — no access keys stored)
      uses: aws-actions/configure-aws-credentials@v4
      with:
        role-to-assume: arn:aws:iam::123456789:role/GitHubActionsRole
        aws-region: ${{ env.AWS_REGION }}

    - name: Run unit tests
      run: |
        cd services/orders
        npm ci && npm test

    - name: Login to ECR
      run: |
        aws ecr get-login-password | docker login \
          --username AWS \
          --password-stdin ${{ env.ECR_REGISTRY }}

    - name: Build Docker image
      run: |
        IMAGE_TAG=${{ github.sha }}
        docker build \
          -t ${{ env.ECR_REGISTRY }}/${{ env.IMAGE_NAME }}:$IMAGE_TAG \
          -t ${{ env.ECR_REGISTRY }}/${{ env.IMAGE_NAME }}:latest \
          --build-arg BUILD_DATE=$(date -u +'%Y-%m-%dT%H:%M:%SZ') \
          --build-arg GIT_SHA=$IMAGE_TAG \
          services/orders/

    - name: Scan image for vulnerabilities (Trivy)
      uses: aquasecurity/trivy-action@master
      with:
        image-ref: ${{ env.ECR_REGISTRY }}/${{ env.IMAGE_NAME }}:${{ github.sha }}
        format: 'sarif'
        exit-code: '1'               # Fail build on CRITICAL/HIGH CVEs
        severity: 'CRITICAL,HIGH'

    - name: Push to ECR
      run: |
        docker push ${{ env.ECR_REGISTRY }}/${{ env.IMAGE_NAME }}:${{ github.sha }}
        docker push ${{ env.ECR_REGISTRY }}/${{ env.IMAGE_NAME }}:latest

    - name: Update image tag in GitOps repo
      run: |
        # Update the Helm values file with the new image tag
        git clone https://x-access-token:${{ secrets.GITOPS_TOKEN }}@github.com/shopnow/k8s-configs.git
        cd k8s-configs
        # Use yq to update the image tag in the Helm values
        yq e ".orders.image.tag = \"${{ github.sha }}\"" \
          -i environments/production/values.yaml
        git config user.email "ci@shopnow.com"
        git config user.name "GitHub Actions"
        git add . && git commit -m "chore: bump orders to ${{ github.sha }}"
        git push
        # ArgoCD detects this git commit and auto-deploys!

    - PruneLast=true       # Delete old resources AFTER new ones are healthy

      backoff:

💡 GitOps principle: Git is the single source of truth. The only way to change production is to merge a PR to the GitOps repo. ArgoCD continuously reconciles K8s state to match Git — if someone manually edits a deployment (kubectl edit), ArgoCD automatically reverts it within 3 minutes. This enforces auditability and repeatability.

apiVersion: monitoring.coreos.com/v1
kind: PrometheusRule
metadata:
  name: shopnow-alerts
  namespace: shopnow
  labels:
    prometheus: kube-prometheus   # Must match Prometheus's ruleSelector labels
spec:
  groups:
  - name: shopnow.orders
    rules:
    - alert: OrdersServiceHighErrorRate
      expr: |
        rate(http_requests_total{
          job="orders-service",
          status=~"5.."
        }[5m]) /
        rate(http_requests_total{
          job="orders-service"
        }[5m]) > 0.05
      for: 2m                     # Fire alert only if condition holds for 2 min
      labels:
        severity: critical
        team: backend
      annotations:
        summary: "Orders service error rate above 5%"
        description: "Error rate is {{ $value | humanizePercentage }} for the last 5 minutes."

    - alert: OrdersServicePodCount
      expr: kube_deployment_status_replicas_available{deployment="orders-service"} < 2
      for: 1m
      labels:
        severity: critical
      annotations:
        summary: "Orders service has fewer than 2 pods running"

Autoscaling — HPA + Cluster Autoscaler + KEDA

Multi-layer autoscaling: pods scale on CPU/RPS, nodes scale on demand, and the SQS consumer scales to zero.

Phase 5

Production Autoscaling Stack

YAMLorders-hpa.yaml
apiVersion: autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: orders-hpa
  namespace: shopnow
spec:
  scaleTargetRef:
    apiVersion: apps/v1
    kind: Deployment
    name: orders-service
  minReplicas: 3
  maxReplicas: 30
  metrics:
  - type: Resource
    resource:
      name: cpu
      target:
        type: Utilization
        averageUtilization: 60
  - type: Pods                         # Custom metric: requests per pod per second
    pods:
      metric:
        name: http_requests_per_second
      target:
        type: AverageValue
        averageValue: "100"            # Scale up if > 100 rps per pod
  behavior:
    scaleUp:
      stabilizationWindowSeconds: 30   # React fast to spikes (Black Friday!)
      policies:
      - type: Percent
        value: 100                     # Can double pod count per 60s
        periodSeconds: 60
    scaleDown:
      stabilizationWindowSeconds: 600  # Wait 10 min before scale down
---
# KEDA: Scale order-processor to ZERO when SQS queue is empty
apiVersion: keda.sh/v1alpha1
kind: ScaledObject
metadata:
  name: order-processor-keda
  namespace: shopnow
spec:
  scaleTargetRef:
    name: order-processor              # Separate consumer Deployment
  minReplicaCount: 0                   # Scale to zero — no orders = no cost
  maxReplicaCount: 40
  cooldownPeriod: 120
  triggers:
  - type: aws-sqs-queue
    metadata:
      queueURL: https://sqs.ap-south-1.amazonaws.com/123456789/shopnow-orders
      queueLength: "50"                # 1 pod per 50 messages
      awsRegion: ap-south-1
    authenticationRef:
      name: keda-aws-auth
💡 Cluster Autoscaler + HPA together: HPA adds pods → pods go Pending (no node capacity) → Cluster Autoscaler detects Pending pods → adds new EC2 node → pods schedule. Scale-down: HPA removes pods → nodes become underutilized → CA safely drains and terminates the node. This is the full auto-scaling loop, from pod to infrastructure.

# ══ STEP 1: What's broken? ══════════════════════════════════════════

🎯 Complete Interview Prep — 50+ Questions

Organized by topic · Designed for 4 YOE Kubernetes / DevOps / Platform Engineering roles

🏗️ Architecture
Q1: Walk me through the Kubernetes control plane. If I run kubectl apply -f deployment.yaml, what happens step by step?
Architecture

1. kubectl reads your kubeconfig, constructs an HTTP POST request, and sends it to the kube-apiserver.

2. The API Server performs authentication (is this a valid user/SA token?), authorization (RBAC — are they allowed to create Deployments?), and runs Admission Controllers (OPA Gatekeeper validates labels; MutatingWebhook injects sidecars).

3. The validated Deployment object is written to etcd.

4. The Deployment Controller in kube-controller-manager watches etcd, notices a new Deployment, and creates a ReplicaSet object.

5. The ReplicaSet Controller notices 0 pods exist vs desired 3, creates 3 Pod objects (Pending state, no node assigned).

6. The Scheduler watches for Pending pods, evaluates nodes (resources, affinity, taints), and assigns each pod to a node by writing the node name to the Pod's spec.nodeName.

7. The kubelet on the assigned node watches its assigned pods, pulls the image from ECR, instructs containerd to start the container, and reports status back.

8. The Endpoint Controller detects the running pod matches the Service selector, adds its IP to the Service's Endpoints. kube-proxy updates iptables rules on all nodes.

🏗️ Architecture
Q2: What happens when an etcd node fails? What about when the whole etcd cluster fails?
Architecture

Single etcd node fails (3-node cluster): etcd uses Raft consensus. With 3 nodes, it can tolerate 1 failure and still form a quorum of 2. The cluster continues operating normally — reads and writes go to the remaining 2 nodes. The failed node can rejoin and sync when it recovers.

Entire etcd cluster fails: The API Server cannot read or write state. kubectl commands fail with timeout errors. Existing pods continue running — kubelet manages containers locally and doesn't stop them because etcd is down. But no new pods can be scheduled, Services can't be updated, and no config changes are possible. Recovery requires restoring etcd from a snapshot backup.

Production setup: Always run 3 or 5 etcd nodes (odd numbers for quorum). On EKS, AWS manages etcd HA. Take scheduled etcd snapshots: etcdctl snapshot save to S3.

📦 Workloads
Q3: What is the difference between a Deployment, StatefulSet, and DaemonSet? Give a real-world use case for each.
Workloads

Deployment: For stateless applications where pods are interchangeable — any pod can handle any request. Use: API servers, web frontends, workers. Supports rolling updates and rollbacks. Pod names have random hashes (orders-7d8f9-abc).

StatefulSet: For applications that need stable network identity, stable persistent storage, and ordered start/stop. Each pod gets a stable ordinal name (mysql-0, mysql-1). Each gets its own PersistentVolume via volumeClaimTemplates. Pods start in order (0→1→2) and stop in reverse (2→1→0). Use: MySQL, PostgreSQL, Kafka, ZooKeeper, Elasticsearch.

DaemonSet: Ensures exactly one pod per node (or selected nodes). Automatically adds pods when nodes join the cluster, removes them when nodes leave. Use: Fluentd (log collection), Prometheus Node Exporter (metrics), AWS VPC CNI (networking), Falco (security monitoring).

📦 Workloads
Q4: Explain PodDisruptionBudget. When is it used and why is it important?
Workloads

A PodDisruptionBudget (PDB) specifies the minimum number (or percentage) of pods that must remain available during voluntary disruptions — node drains, Cluster Autoscaler scale-down, or rolling upgrades.

Without a PDB: when you run kubectl drain for node maintenance, Kubernetes may evict all pods from a deployment simultaneously, causing a full outage. With minAvailable: 2: only 1 pod can be disrupted at a time — the drain proceeds node by node, keeping at least 2 pods running throughout.

When it's used: Node maintenance windows, Kubernetes version upgrades (rolling node replacement), Cluster Autoscaler removing underutilized nodes, kube-scheduler rebalancing. Always set a PDB for production services with 2+ replicas.

📦 Workloads
Q5: What is the purpose of topologySpreadConstraints? How is it different from podAntiAffinity?
Workloads

podAntiAffinity prevents two pods from being on the same node or in the same zone — but doesn't care about balance. With 6 pods and 3 zones, you could get 5 in zone-A and 1 in zone-B.

topologySpreadConstraints ensures even distribution. maxSkew: 1 means the difference between the most and least loaded topology domain (zone or node) cannot exceed 1. With 6 pods and 3 zones, it enforces 2-2-2 distribution. This prevents a zone failure from taking out a disproportionate number of pods. Use both together: antiAffinity for node-level separation, topologySpreadConstraints for zone-level balance.

🌐 Networking
Q6: Explain how a request flows from the internet to a pod inside EKS. Include every layer.
Networking

1. User's browser sends HTTPS GET https://shopnow.com/api/orders.

2. Route 53 resolves shopnow.com to the ALB DNS name.

3. AWS ALB (created by AWS Load Balancer Controller from the Ingress resource): terminates TLS (ACM cert), optionally runs WAF rules, and forwards HTTP to the Ingress Controller target group.

4. Ingress Controller pod (nginx running in ingress-nginx namespace): reads Ingress rules, matches /api path → forwards to api-gateway-svc:8080.

5. Service (api-gateway-svc, ClusterIP): kube-proxy's iptables rules on the node intercept the ClusterIP, select a pod endpoint at random (load balancing), and rewrite the destination IP to the pod's VPC IP.

6. AWS VPC CNI routes the packet directly to the pod's VPC IP (no overlay, no NAT) — the pod is a real VPC IP.

7. The api-gateway pod receives the request, processes it, and may call downstream services by their ClusterIP (e.g., orders-service.shopnow.svc.cluster.local:3000) resolved via CoreDNS.

🌐 Networking
Q7: What are the different types of Services in Kubernetes? When would you use each?
Networking

ClusterIP (default): Gives a stable virtual IP accessible only within the cluster. Use for all internal service-to-service communication. Example: API calling the Orders service.

NodePort: Exposes the service on a static port (30000-32767) on every node. Accessible from outside the cluster via any node IP. Use only for development/testing — not production (no SSL, no path routing, node IP is not stable).

LoadBalancer: Provisions a cloud load balancer (NLB on EKS by default with AWS LBC). Use for services that need direct external access without Ingress — e.g., a TCP service, a game server, or a VPN endpoint. Each service gets its own LB ($).

ExternalName: Maps a K8s service name to an external DNS name. Pods call rds-svc, which resolves to mydb.abc123.ap-south-1.rds.amazonaws.com. Use for integrating external services into cluster DNS without hardcoding external URLs in code.

Headless (clusterIP: None): No virtual IP — DNS returns individual pod IPs directly. Required by StatefulSets for stable pod DNS names. Use for databases, where clients need to connect to a specific replica.

🌐 Networking
Q8: How do Network Policies work? What CNI is required, and what is the default behavior without them?
Networking

Default: All pods can communicate with all pods across all namespaces — completely open flat network. A compromised frontend pod could directly connect to your database.

Network Policies are firewall rules for pods. They select pods (via podSelector) and define allowed ingress (incoming) and egress (outgoing) traffic. Once you apply any ingress NetworkPolicy to a pod, only traffic matching the rules is allowed — everything else is denied (implicit deny).

CNI requirement: NetworkPolicies are implemented by the CNI plugin, not by core Kubernetes. AWS VPC CNI alone does NOT enforce them — you need the Amazon VPC CNI Network Policy Controller add-on (eks addon: vpc-cni with enableNetworkPolicy=true) or a third-party CNI like Calico or Cilium.

Best practice: Start with a deny-all ingress policy in every namespace, then add explicit allow rules for only the required communication paths.

💾 Storage
Q9: A pod is stuck in Pending state and shows "0/3 nodes are available: pod has unbound immediate PersistentVolumeClaims." How do you fix this?
Storage

This means the PVC is in Pending state — no PV has been bound to it. Steps to diagnose:

1. kubectl get pvc -n shopnow — confirm PVC is Pending.

2. kubectl describe pvc PVC_NAME -n shopnow — check Events for the root cause.

Common causes and fixes: StorageClass not found → typo in storageClassName, or the EBS CSI driver isn't installed. WaitForFirstConsumer → PVC remains Pending until the pod is scheduled to a node — this is normal and expected. No nodes in the right AZ → EBS is AZ-specific; if all nodes are in 1a but the EBS was provisioned in 1b, it can't attach. Fix: use volumeBindingMode: WaitForFirstConsumer on the StorageClass. Capacity → AWS account EBS limit reached. Check Service Quotas in AWS console.

💾 Storage
Q10: What happens to your data when you delete a PVC with reclaimPolicy: Delete vs Retain?
Storage

reclaimPolicy: Delete: When the PVC is deleted, Kubernetes automatically deletes the PV and the underlying EBS volume. Data is permanently lost. This is the default for dynamically provisioned volumes in most StorageClasses. Use for dev/test environments.

reclaimPolicy: Retain: When the PVC is deleted, the PV remains (status: Released) and the EBS volume is preserved. An admin must manually delete the PV and EBS volume, or reclaim the PV for a new PVC. Data is safe. Use for production databases.

Critical note: Check the StorageClass's reclaimPolicy BEFORE using it in production — many default StorageClasses use Delete. Always use Retain for StatefulSet databases in production.

🔐 Security & RBAC
Q11: How would you give a new CI/CD system permission to deploy to EKS, following least-privilege principles?
Security

1. Create a dedicated Kubernetes ServiceAccount in the target namespace: kubectl create sa cicd-deployer -n shopnow.

2. Create a Role with minimal permissions — only what CI/CD needs: get/list/update Deployments, get/list Pods, get/list/create ConfigMaps. Do NOT grant access to Secrets, Nodes, or other namespaces.

3. Create a RoleBinding to bind the Role to the ServiceAccount.

4. On GitHub Actions: use OIDC (OpenID Connect) — no long-lived AWS access keys. GitHub proves its identity to AWS, assumes an IAM role, and gets temporary credentials. The IAM role's trust policy restricts it to only your specific repo and branch.

5. Use eksctl create iamserviceaccount or AWS IAM to link the GitHub OIDC identity to an IAM role that has EKS eks:DescribeCluster permission and is mapped in aws-auth to the cicd-deployer ServiceAccount's permissions.

🔐 Security & RBAC
Q12: What is the difference between PodSecurityPolicy (deprecated), Pod Security Admission (PSA), and OPA Gatekeeper?
Security

PodSecurityPolicy (PSP): Deprecated in K8s 1.21, removed in 1.25. Cluster-wide policies to control pod security settings. Complex to configure — many clusters had PSP disabled or misconfigured.

Pod Security Admission (PSA): Built-in replacement (K8s 1.25+). Applied via namespace labels. Enforces one of three profiles (privileged, baseline, restricted) per namespace. Simple to adopt — just add labels. Limitation: only three fixed profiles, no custom rules.

OPA Gatekeeper / Kyverno: Webhook-based policy engines. Fully customizable — you write policies in Rego (Gatekeeper) or YAML (Kyverno) to enforce anything: require specific labels, block certain images, enforce resource limits, restrict hostPath. More powerful than PSA but requires installation and maintenance.

Production recommendation: Use PSA for baseline pod security across all namespaces. Add Kyverno or Gatekeeper for custom organizational policies (require team labels, block latest image tag, enforce resource limits).

🔧 Operations & Troubleshooting
Q13: How do you perform a zero-downtime rolling update in Kubernetes? What settings make it truly zero-downtime?
Operations

A Deployment's RollingUpdate strategy replaces pods gradually. But for true zero-downtime you need all of these configured:

1. maxUnavailable: 0 — never reduce the pod count below desired during update. Ensures minimum capacity is always met.

2. maxSurge: 1 — temporarily run one extra pod during the update so traffic is always served.

3. readinessProbe — new pod only receives traffic after the probe passes. Without this, K8s routes traffic to pods before the app has finished starting.

4. lifecycle.preStop — delay SIGTERM by 15-30s to allow in-flight requests to complete and the load balancer to stop routing to the old pod.

5. terminationGracePeriodSeconds — set longer than the longest expected request (e.g., 60s). After this period, SIGKILL is sent.

6. PodDisruptionBudget — prevents accidental disruption during simultaneous cluster maintenance.

🔧 Operations & Troubleshooting
Q14: A pod shows CrashLoopBackOff. Walk me through your debugging process step by step.
Debugging

1. kubectl logs POD_NAME -n shopnow --previous — see crash output from the last container run. This is usually the most revealing step.

2. kubectl describe pod POD_NAME -n shopnow — check Events section for image pull issues, liveness probe failures, OOM kills, or resource limit errors.

3. If no useful logs: kubectl get events -n shopnow --sort-by=.lastTimestamp — cluster-level events may show the error.

4. Check for OOMKilled in the describe output under Last State. If yes: increase resources.limits.memory.

5. Check if environment variables are set correctly — a missing DB_HOST or wrong SECRET reference causes crashes at startup.

6. If the app starts but crashes later: check liveness probe path and timing. A too-aggressive liveness probe restarts healthy pods.

7. Temporarily add command: ["sleep","3600"] to the container spec to prevent the crash and get a shell inside: kubectl exec -it POD -- /bin/sh. Manually run the app command to see the real error.

🔧 Operations & Troubleshooting
Q15: How does the Cluster Autoscaler decide to add or remove nodes?
Operations

Scale Up trigger: A pod is stuck in Pending state because no existing node has enough CPU, memory, or the correct labels/tolerations. CA simulates placing the Pending pod on a node from each node group and chooses the most cost-efficient group that can accommodate it. It adds nodes from that group's EC2 Auto Scaling Group. This typically takes 2-4 minutes (EC2 boot time).

Scale Down trigger: A node's CPU and Memory are both below 50% utilization for more than 10 minutes (configurable). CA checks if all pods on that node can be safely rescheduled to other nodes. If yes, it drains the node (evicts pods, respecting PDBs) and terminates the EC2 instance.

Scale Down blockers: Pods with local storage (hostPath, emptyDir with data), pods with no controller (standalone pods), pods with PDB that would be violated, pods with cluster-autoscaler.kubernetes.io/safe-to-evict: "false" annotation.

🔧 Operations & Troubleshooting
Q16: How do you upgrade EKS cluster version with zero downtime?
Operations

1. Check compatibility: Run kubectl get nodes to see current versions. Read EKS changelog for breaking changes. Check add-ons compatibility matrix.

2. Upgrade control plane: aws eks update-cluster-version --name shopnow-prod --kubernetes-version 1.30. This takes ~15 minutes. EKS does a rolling control plane upgrade with HA — API Server stays accessible throughout.

3. Upgrade add-ons: aws eks update-addon for vpc-cni, coredns, kube-proxy. They must be version-compatible with the new K8s version.

4. Upgrade node groups: Use managed node groups — aws eks update-nodegroup-version. This replaces nodes one at a time: launches new node, drains old node (respecting PDBs), terminates old node. Your workloads reschedule to new nodes. Zero downtime because PDBs ensure minimum pod availability.

5. Test thoroughly in a staging cluster one version ahead of production before upgrading prod.

🎭 Scenario-Based
Q17: It's Black Friday. Traffic is 10x normal. The orders service HPA is at maxReplicas=30 but pods are still failing. What's your diagnosis and response?
Scenario

1. HPA at max but pods failing: First check — are pods Pending or Running-but-overloaded? kubectl get pods -n shopnow | grep orders.

2. If Pending: Cluster Autoscaler hasn't added nodes fast enough. Check kubectl describe pod PENDING_POD for "Insufficient CPU/memory". Fix: Pre-scale nodes before Black Friday (eksctl scale nodegroup --nodes=20). Or increase CA's scan interval. Or use Spot instance warm pools.

3. If Running but overloaded: The bottleneck isn't the orders service — it's a dependency. Check: PostgreSQL connection pool exhausted? Redis hitting memory limit? Downstream payment service has no HPA? Each service in the chain needs its own autoscaling.

4. Immediate mitigation: Manually increase maxReplicas temporarily: kubectl patch hpa orders-hpa -p '{"spec":{"maxReplicas":50}}'. Enable pod rate limiting or circuit breakers. Serve cached responses for read-heavy endpoints.

5. Post-incident: Load test with realistic Black Friday traffic before the event. Use KEDA for event-driven scaling. Pre-provision nodes using scheduled scaling in the Auto Scaling Group.

🎭 Scenario-Based
Q18: You find that a developer has been running kubectl exec into production pods and manually editing config files. How do you prevent this and enforce proper change management?
Scenario + Security

1. RBAC fix: Remove pods/exec from developer Role in the production namespace. Only SRE/on-call roles should have exec in production: verbs: ["get","list","watch"] for devs.

2. Immutable pods: readOnlyRootFilesystem: true in SecurityContext — even if someone exec's in, they can't write to the filesystem.

3. GitOps enforcement: ArgoCD with selfHeal: true automatically reverts any manual kubectl changes within 3 minutes — it makes manual edits pointless.

4. Audit logging: Enable EKS API Server audit logs to CloudWatch. Set a CloudWatch alarm for pods/exec API calls in production. All exec sessions are logged with the IAM identity of the user.

5. Process: Require all config changes as PRs to the GitOps repo. ArgoCD deploys approved changes. Runbooks for emergencies that require exec access need a documented approval process.

🎭 Scenario-Based
Q19: Design a disaster recovery strategy for a stateful application (PostgreSQL) running on EKS.
Scenario + Storage

RTO/RPO targets: First agree on Recovery Time Objective (how long can the app be down) and Recovery Point Objective (how much data loss is acceptable). This drives the strategy.

Layer 1 — EBS Snapshots (RPO: hours): Use the EBS CSI Driver's VolumeSnapshot to take daily/hourly snapshots of PostgreSQL PVCs. Snapshots are stored in S3 (cross-AZ). Restore: create a new PVC from the snapshot.

Layer 2 — PostgreSQL WAL archiving (RPO: minutes): Configure PostgreSQL to continuously ship Write-Ahead Logs to S3 using pg_basebackup + WAL-G or Barman. Restore to any point-in-time by replaying WAL files.

Layer 3 — Read replica (RPO: seconds): Run a PostgreSQL StatefulSet with 1 primary + 1 replica in a different AZ. If the primary AZ fails, promote the replica. The pods run in anti-affinity across AZs.

Layer 4 — Cross-region DR (RPO: minutes, RTO: ~30min): Replicate EBS snapshots to a second region using Data Lifecycle Manager. Maintain a second EKS cluster in ap-southeast-1 with infrastructure ready but scaled to zero. Activate it by restoring snapshots and scaling pods.

🔧 Operations
Q20: What is the difference between Horizontal Pod Autoscaler, Vertical Pod Autoscaler, and Cluster Autoscaler? How do they interact?
Scaling

HPA (Horizontal Pod Autoscaler): Scales the NUMBER of pods. Adds/removes replicas based on CPU, memory, or custom metrics (RPS, queue depth). Works at the workload level. Fast — pod scheduling takes seconds.

VPA (Vertical Pod Autoscaler): Scales the SIZE of pods — adjusts requests and limits. Use for apps that can't run multiple replicas (single-instance DB) or to right-size resource requests. Pod must restart to apply changes (Auto mode evicts pod).

Cluster Autoscaler: Scales the NUMBER of EC2 NODES. Adds nodes when pods are Pending due to resource shortage. Removes underutilized nodes. Works at the infrastructure level. Slow — takes 2-5 minutes (EC2 boot).

Interaction: HPA tries to add pods → not enough node capacity → pods go Pending → CA adds a node → pods schedule. Scale down: HPA removes pods → nodes underutilized → CA drains and removes nodes. The chain is: load → HPA → pod count change → CA → node count change. Never use HPA and VPA on the same resource metric (conflict). Combine: HPA for CPU/RPS, KEDA for queue depth, CA for node provisioning.
