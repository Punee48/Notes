🔧 Lab — Jenkinsfile (Build + Test + Docker Push)
Jenkinsfile — Part 1: Build and Docker push stages
// Jenkinsfile — TechNova Full CI/CD Pipeline
// Declarative Pipeline syntax (cleaner than scripted)

pipeline {

  // "agent any" = run this pipeline on any available Jenkins agent
  // For production: agent { label 'docker-agent' } — specific agent type
  agent any

  // "environment" block defines variables available to ALL stages
  // credentials() fetches secret from Jenkins Credentials Store (never hardcoded)
  environment {
    APP_NAME        = 'technova-app'
    ECR_REGISTRY    = '123456789.dkr.ecr.ap-south-1.amazonaws.com'
    ECR_REPO        = "${ECR_REGISTRY}/technova-app"
    AWS_REGION      = 'ap-south-1'
    // credentials('aws-ecr-credentials') = Jenkins Credential ID
    // Jenkins injects AWS_ACCESS_KEY_ID and AWS_SECRET_ACCESS_KEY automatically
    AWS_CREDENTIALS = credentials('aws-ecr-credentials')
    ANSIBLE_PRIVATE_KEY = credentials('technova-ssh-key')
    // ANSIBLE_PRIVATE_KEY = the .pem file stored as Jenkins "SSH Username + key" credential
    TOWER_TOKEN     = credentials('ansible-tower-token')
    TOWER_HOST      = 'https://tower.technova.com'
    TOWER_TEMPLATE_ID = '5'
  }

  // "options" — global pipeline behaviour settings
  options {
    buildDiscarder(logRotator(numToKeepStr: '20'))
    // Keep only last 20 build logs — don't fill up Jenkins disk
    timeout(time: 45, unit: 'MINUTES')
    // Abort if entire pipeline takes longer than 45 minutes
    disableConcurrentBuilds()
    // Don't run two builds of the same pipeline simultaneously
    // Prevents race conditions in deployments
  }

  stages {

    // ── STAGE 1: CHECKOUT ──────────────────────────────────
    stage('Checkout') {
      steps {
        checkout scm
        // "checkout scm" = clone the Git repo that triggered this build
        // Ansible playbooks are in a subdirectory: ansible/
        script {
          // Capture Git commit SHA for Docker image tag
          env.GIT_COMMIT_SHORT = sh(
            script: 'git rev-parse --short HEAD',
            returnStdout: true
          ).trim()
          // returnStdout: true = capture command output as a string
          // .trim() = remove trailing newline
          env.APP_VERSION = env.GIT_COMMIT_SHORT
          echo "Building version: ${env.APP_VERSION}"
        }
      }
    }

    // ── STAGE 2: BUILD ─────────────────────────────────────
    stage('Build') {
      steps {
        sh """
          echo "Building TechNova App version ${APP_VERSION}"
          mvn clean package -DskipTests -q
          # -DskipTests : skip tests here — we run them separately
          # -q          : quiet mode, less verbose output
        """
      }
      post {
        success {
          archiveArtifacts artifacts: 'target/*.jar', fingerprint: true
          // Save the built JAR as a Jenkins artifact for traceability
        }
      }
    }

    // ── STAGE 3: UNIT TESTS ────────────────────────────────
    stage('Unit Tests') {
      steps {
        sh 'mvn test -q'
      }
      post {
        always {
          // "always" runs even if tests fail — so reports are published
          junit 'target/surefire-reports/*.xml'
          // Publishes JUnit XML results to Jenkins test dashboard
        }
        failure {
          echo "❌ Unit tests failed — stopping pipeline"
          // When tests fail → stage fails → pipeline stops automatically
          // Subsequent stages (Docker build, Ansible deploy) do NOT run
        }
      }
    }

    // ── STAGE 4: BUILD & PUSH DOCKER IMAGE ─────────────────
    stage('Docker Build & Push to ECR') {
      steps {
        sh """
          # Login to AWS ECR
          aws ecr get-login-password --region ${AWS_REGION} \
            | docker login --username AWS \
              --password-stdin ${ECR_REGISTRY}

          # Build image with TWO tags:
          # 1. git commit SHA tag (immutable, traceable)
          # 2. "latest" tag (always points to newest)
          docker build \
            --tag ${ECR_REPO}:${APP_VERSION} \
            --tag ${ECR_REPO}:latest \
            --build-arg APP_VERSION=${APP_VERSION} \
            .

          # Push BOTH tags to ECR
          docker push ${ECR_REPO}:${APP_VERSION}
          docker push ${ECR_REPO}:latest

          echo "✅ Pushed ${ECR_REPO}:${APP_VERSION} to ECR"
        """
      }
    }
  }

  // "post" block runs after ALL stages regardless of outcome
  post {
    always {
      // Clean Docker images from Jenkins agent to save disk space
      sh 'docker system prune -f --filter "until=24h" || true'
      // "|| true" prevents prune failure from failing the build
    }
  }
}

// Add these stages AFTER the Docker Build & Push stage above

    // ── STAGE 5: DEPLOY TO STAGING ─────────────────────────
    stage('Deploy to Staging') {
      steps {
        // sshagent adds the SSH key to the agent for Ansible to use
        sshagent(credentials: ['technova-ssh-key']) {
          sh """
            cd ansible/   # Playbooks are in the ansible/ subdirectory

            # Run Ansible playbook targeting staging inventory
            # "-i" : inventory file (staging servers only)
            # "--extra-vars" : override variables at runtime
            # "app_version=${APP_VERSION}" : the exact git SHA to deploy
            ansible-playbook \
              -i inventory/staging.ini \
              playbooks/deploy.yml \
              --extra-vars "app_version=${APP_VERSION} \
                            docker_image=${ECR_REPO}:${APP_VERSION} \
                            app_environment=staging" \
              -v
          """
        }
      }
    }

    // ── STAGE 6: SMOKE TESTS ON STAGING ────────────────────
    stage('Smoke Tests — Staging') {
      steps {
        sh """
          # Run smoke tests against staging environment
          # These are quick HTTP-based tests to verify basic functionality
          python3 tests/smoke_test.py --env staging --url https://staging.technova.com
        """
      }
    }

    // ── STAGE 7: PRODUCTION DEPLOY (only on main branch) ───
    stage('Deploy to Production') {
      // "when" condition: only run this stage when on 'main' branch
      // Feature branches deploy to staging only — never production
      when { branch 'main' }

      steps {
        // "input" pauses the pipeline and waits for a human to approve
        // Shows a dialog box in the Jenkins UI: "Deploy to Production?"
        // If nobody approves within 30 min, the pipeline times out and fails
        input message: 'Deploy to Production?',
              ok: 'Yes, Deploy',
              parameters: [
                string(name: 'REASON', description: 'Reason for deployment')
              ]

        sshagent(credentials: ['technova-ssh-key']) {
          sh """
            cd ansible/

            # Production deploy uses dynamic inventory (aws_ec2 plugin)
            # serial=5 : rolling deploy 5 servers at a time
            ansible-playbook \
              -i inventory/aws_ec2.yml \
              playbooks/deploy.yml \
              --extra-vars "app_version=${APP_VERSION} \
                            docker_image=${ECR_REPO}:${APP_VERSION} \
                            app_environment=production \
                            serial_count=5" \
              --limit env_production \
              -v
          """
        }
      }

      post {
        success {
          // Tag this Git commit as a production release
          sh """
            git tag -a "prod-${APP_VERSION}" \
              -m "Deployed to production by Jenkins build ${BUILD_NUMBER}"
            git push origin "prod-${APP_VERSION}"
          """
        }
      }
    }

  } // end stages

  post {
    success {
      slackSend(
        channel: '#deployments',
        color:   'good',
        message: """✅ *TechNova Deploy SUCCESS*
Version: `${APP_VERSION}`
Branch:  `${GIT_BRANCH}`
Build:   #${BUILD_NUMBER}
By:      ${currentBuild.getBuildCauses()[0].userId ?: 'scheduled'}
View:    ${BUILD_URL}"""
      )
    }
    failure {
      slackSend(
        channel: '#deployments',
        color:   'danger',
        message: """❌ *TechNova Deploy FAILED*
Version: `${APP_VERSION}`
Stage:   ${currentBuild.result}
Build:   #${BUILD_NUMBER}
Logs:    ${BUILD_URL}console"""
      )
    }
    always {
      sh 'docker system prune -f --filter "until=24h" || true'
      cleanWs()
      // cleanWs() = clean workspace after build — saves Jenkins disk space
    }
  }
}

Option B — Jenkins Triggers Tower Instead of Direct Ansible

🧠 Direct ansible-playbook vs Tower API — When to Use Which

Both approaches work. Direct ansible-playbook: simpler, works without Tower, Jenkins itself shows the output. Use for smaller teams without Tower. Tower API: enterprise approach — Tower provides RBAC, full audit trail separate from Jenkins, Tower handles credentials securely, developers can re-trigger from Tower UI without Jenkins. Use when your company has Tower.

Jenkins stage using Tower API instead of direct ansible-playbook
stage('Deploy via Ansible Tower') {
  when { branch 'main' }
  steps {
    script {
      // Step 1: Launch Tower Job Template via REST API
      def resp = sh(script: """
        curl -s -X POST ${TOWER_HOST}/api/v2/job_templates/${TOWER_TEMPLATE_ID}/launch/ \
          -H "Authorization: Bearer ${TOWER_TOKEN}" \
          -H "Content-Type: application/json" \
          -d '{
            "extra_vars": {
              "app_version": "${APP_VERSION}",
              "docker_image": "${ECR_REPO}:${APP_VERSION}",
              "app_environment": "production"
            }
          }'
      """, returnStdout: true).trim()

      // Step 2: Extract job ID from response
      def jobId = sh(
        script: """echo '${resp}' | python3 -c "import sys,json; print(json.load(sys.stdin)['id'])" """,
        returnStdout: true
      ).trim()

      echo "Tower Job ID: ${jobId}"
      echo "Monitor at: ${TOWER_HOST}/#/jobs/${jobId}"

      // Step 3: Poll until job finishes (max 20 minutes)
      def maxRetries = 60; def retryCount = 0
      while (retryCount < maxRetries) {
        sleep 20
        retryCount++
        def status = sh(script: """
          curl -s ${TOWER_HOST}/api/v2/jobs/${jobId}/ \
            -H "Authorization: Bearer ${TOWER_TOKEN}" \
            | python3 -c "import sys,json; print(json.load(sys.stdin)['status'])"
        """, returnStdout: true).trim()

        echo "[${retryCount * 20}s] Status: ${status}"

        if (status == 'successful') {
          echo "✅ Tower deployment SUCCESSFUL"; return
        } else if (status in ['failed', 'error', 'canceled']) {
          error "❌ Tower deployment FAILED: ${status}"
        }
      }
      error "Timeout: Tower job did not complete in 20 minutes"
    }
  }
}
🎯 Interview Questions — Chapter 23
Q1: How do you pass SSH credentials to Ansible inside a Jenkins pipeline securely?
Store the SSH private key as a Jenkins "SSH Username with private key" Credential. In the Jenkinsfile, wrap the Ansible step with sshagent(credentials: ['credential-id']). This starts an SSH agent process that holds the key in memory for the duration of that block. Ansible's SSH connections automatically use the key from the agent — the key is never written to disk or visible in logs. Never store SSH keys as environment variables or in the code itself.
Q2: What is the "when" block in a Jenkinsfile and how do you use it for branch-based deployments?
The when block defines conditions under which a stage runs. when { branch 'main' } means the stage only executes when the build is triggered by a push to the main branch. Feature branches run all previous stages (build, test, Docker push) but skip the production deploy stage. This prevents accidental production deployments from feature branches while still validating code quality. You can also use when { environment name: 'DEPLOY_ENV', value: 'prod' } or when { expression { return params.DEPLOY_TO_PROD } }.
Q3: What does the "input" step do in a Jenkinsfile?
The input step pauses the pipeline and waits for a human to click Approve or Abort in the Jenkins UI (or via API). It's used as a manual gate before critical deployments — like a production deploy. The pipeline shows a dialog box: "Deploy to Production? Yes/No". If approved, the pipeline continues; if aborted or if nobody responds within the timeout, the build is marked as aborted. This is the Jenkins equivalent of Tower's Workflow Approval node — a human checkpoint in an otherwise automated pipeline.

💡 Simple Mental Model

Terraform = Create and manage cloud resources (things in the AWS console). Ansible = Configure the operating system and software inside those resources (what you'd do if you SSHed in). If you can click it in the AWS console, Terraform manages it. If you'd SSH in and type a command, Ansible manages it.

Pattern 1 — Terraform Triggers Ansible After EC2 Creation

🧠 local-exec Provisioner

Terraform's local-exec provisioner runs a command on your local machine (not the remote server) after a resource is created. You can use this to trigger an Ansible playbook immediately after Terraform creates an EC2 instance — passing the new instance's IP automatically.

🔧 Lab — Terraform + Ansible via local-exec
main.tf — EC2 instance with Ansible provisioner
# main.tf — TechNova Singapore Region Deployment

terraform {
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
  # Remote state so team can share state file
  backend "s3" {
    bucket = "technova-terraform-state"
    key    = "singapore/terraform.tfstate"
    region = "ap-south-1"
  }
}

provider "aws" {
  region = "ap-southeast-1"   # Singapore
}

# ── Variables ───────────────────────────────────────────
variable "key_name"       { default = "technova-sg-key" }
variable "instance_type"  { default = "t3.medium" }
variable "web_count"      { default = 3 }
variable "app_version"    { default = "latest" }

# ── EC2 Instances for Web Tier ───────────────────────────
resource "aws_instance" "web" {
  count         = var.web_count
    # Create "web_count" identical instances
  ami           = data.aws_ami.ubuntu.id
  instance_type = var.instance_type
  key_name      = var.key_name
  subnet_id     = aws_subnet.private[count.index % length(aws_subnet.private)].id
  vpc_security_group_ids = [aws_security_group.web_sg.id]

  tags = {
    Name        = "technova-web-${count.index + 1}"
    Project     = "TechNova"
    Environment = "production"
    Role        = "webserver"
    Managed     = "ansible"   # Tag required by Ansible dynamic inventory
  }

    # local-exec: runs ON YOUR LOCAL MACHINE after this EC2 is created
    # self.public_ip = the new EC2's public IP (Terraform fills this in)
  provisioner "local-exec" {
    command = <<-EOF
      echo "Waiting 60 seconds for SSH to be ready..."
      sleep 60

      echo "Running Ansible against ${self.public_ip}..."
      ansible-playbook \
        -i "${self.public_ip}," \
        -u ubuntu \
        --private-key ~/.ssh/technova-sg-key.pem \
        --extra-vars "app_version=${var.app_version}" \
        ansible/playbooks/site.yml \
        -v
    EOF
        # "${self.public_ip}," — the trailing comma makes it a valid inventory string
        # Ansible accepts "10.0.1.5," as a single-host inventory
  }

    # depends_on = only run Ansible AFTER the security group exists
  depends_on = [aws_security_group.web_sg]
}

# ── RDS Database ─────────────────────────────────────────
resource "aws_db_instance" "main" {
  identifier     = "technova-sg-prod"
  engine         = "mysql"
  engine_version = "8.0"
  instance_class = "db.t3.medium"
  allocated_storage = 100
  db_name        = "technova_db"
  username       = "admin"
  password       = var.db_password   # from terraform.tfvars (gitignored)
  multi_az       = true
  skip_final_snapshot = false
  final_snapshot_identifier = "technova-final-snap"
  tags = { Project = "TechNova", Environment = "production" }
}

output "web_instance_ips" {
  value = aws_instance.web[*].public_ip
    # [*] = spread operator, gets all public IPs from the count loop
}

output "db_endpoint" {
  value     = aws_db_instance.main.endpoint

Pattern 2 — Terraform Output → Ansible Dynamic Inventory

🧠 The Better Pattern for Production

The local-exec approach is simple but fragile — if Ansible fails, Terraform marks the resource as tainted. A cleaner production approach: Terraform applies first (creates all infrastructure), then a separate Ansible run configures everything using EC2 tags for dynamic inventory. The two tools are completely decoupled.

🔧 Lab — Two-Step Pipeline: Terraform → Ansible
deploy-singapore.sh — the complete deployment script
#!/bin/bash
# deploy-singapore.sh — Provision + Configure TechNova Singapore
# Usage: ./deploy-singapore.sh 2.6.0

APP_VERSION=${1:-latest}
echo "🚀 Deploying TechNova Singapore — Version: $APP_VERSION"

# ── PHASE 1: TERRAFORM — Create Infrastructure ────────────
echo "📐 Phase 1: Provisioning infrastructure with Terraform..."

cd terraform/singapore/

# Initialize Terraform (download providers, setup backend)
terraform init

# Plan — show what will be created (good for review)
terraform plan \
  -var="app_version=$APP_VERSION" \
  -out=singapore.tfplan
  # -out saves the plan to a file — apply will use this exact plan
  # Prevents changes between plan and apply (important in CI/CD)

# Apply the saved plan — creates EC2s, RDS, VPC, ALB, SG...
terraform apply singapore.tfplan

# Get Terraform outputs as JSON for use in Ansible
DB_ENDPOINT=$(terraform output -raw db_endpoint)
WEB_IPS=$(terraform output -json web_instance_ips)
echo "DB Endpoint: $DB_ENDPOINT"
echo "Web IPs    : $WEB_IPS"

cd ../../

# ── PHASE 2: WAIT — SSH takes ~60 seconds after EC2 launch ─
echo "⏳ Waiting 90 seconds for SSH to become available..."
sleep 90

# ── PHASE 3: ANSIBLE — Configure & Deploy ─────────────────
echo "🤖 Phase 3: Configuring servers with Ansible..."

cd ansible/

# Use aws_ec2 dynamic inventory — picks up ALL new instances by tag
# The EC2 instances were tagged Role=webserver by Terraform
# aws_ec2.yml groups them into "role_webserver" automatically
ansible-playbook \
  -i inventory/aws_ec2_singapore.yml \
  playbooks/site.yml \
  --limit role_webserver \
  --extra-vars "app_version=$APP_VERSION db_endpoint=$DB_ENDPOINT" \
  -v

echo "✅ TechNova Singapore is LIVE!"
echo "   - $(terraform -chdir=../terraform/singapore output instance_count) web servers"
echo "   - Version: $APP_VERSION"

Pattern 3 — null_resource for Re-running Ansible

null_resource — run Ansible from Terraform on-demand
# null_resource has no cloud resource — it's just a hook for provisioners
# Useful when you want to re-run Ansible without re-creating EC2 instances

resource "null_resource" "configure_webservers" {

    # "triggers" define when this resource re-runs
    # If any trigger value changes → null_resource is destroyed + re-created
    # → local-exec runs again → Ansible re-runs
  triggers = {
    app_version    = var.app_version
        # Re-run Ansible when app_version changes
    instance_ids   = join(",", aws_instance.web[*].id)
        # Re-run if any EC2 instance ID changes (new instance added)
    playbook_hash  = filemd5("${path.module}/../ansible/playbooks/site.yml")
        # Re-run if the playbook file itself changes — smart!
        # filemd5() calculates MD5 hash of the file
  }

    # depends_on: wait for ALL EC2 instances to be created first
  depends_on = [aws_instance.web]

  provisioner "local-exec" {
    command = <<-EOF
      sleep 60  # Wait for SSH

      # Write a temporary inventory file from Terraform's known IPs
      # More reliable than waiting for dynamic inventory to catch up
      echo "[webservers]" > /tmp/tf_inventory.ini
      echo "${join("\n", aws_instance.web[*].public_ip)}" >> /tmp/tf_inventory.ini
      echo "[webservers:vars]" >> /tmp/tf_inventory.ini
      echo "ansible_user=ubuntu" >> /tmp/tf_inventory.ini
      echo "db_endpoint=${aws_db_instance.main.endpoint}" >> /tmp/tf_inventory.ini

      ansible-playbook \
        -i /tmp/tf_inventory.ini \
        --private-key ~/.ssh/technova-sg-key.pem \
        --extra-vars "app_version=${var.app_version}" \
        ansible/playbooks/site.yml

      rm /tmp/tf_inventory.ini  # Clean up temp file
    EOF
  }
}
🎯 Interview Questions — Chapter 24
Q1: How do Terraform and Ansible complement each other? Where is the boundary?
They have a clean separation: Terraform manages cloud infrastructure — VPCs, EC2 instances, RDS, ALB, S3, IAM roles — anything you create in the AWS console. It tracks state and can create, update, or destroy cloud resources. Ansible manages OS-level configuration — what you'd do if you SSHed into a server: installing packages, configuring files, managing services, deploying applications. Terraform creates the EC2 instance; Ansible configures what runs on it. Terraform is declarative and state-tracked for cloud APIs; Ansible is procedural and SSH-based for server configuration.
Q2: What is the local-exec provisioner in Terraform and what is its limitation?
The local-exec provisioner runs a shell command on your local machine (the machine running terraform) after a resource is created. It's used to trigger Ansible after EC2 creation: pass the new IP, run the playbook. Limitation: if the provisioner fails (Ansible fails), Terraform marks the resource as "tainted" — it will try to destroy and recreate the EC2 on the next apply. This couples infrastructure creation with configuration. The cleaner approach for production: use Terraform for infrastructure only, then run Ansible as a separate step in your CI/CD pipeline using EC2 tags and dynamic inventory, completely decoupled.
Q3: How do you pass Terraform outputs (like RDS endpoint) to Ansible?
Several approaches: (1) Command line extra_vars: capture with terraform output -raw db_endpoint in a shell variable and pass with --extra-vars "db_endpoint=$DB_ENDPOINT". (2) Terraform writes a vars file: use local_file resource to write a YAML file that Ansible reads with vars_files:. (3) Environment variables: export as env vars, Ansible's lookup('env', 'DB_ENDPOINT') reads them. (4) AWS SSM Parameter Store: Terraform writes outputs to SSM, Ansible reads them at runtime with the aws_ssm lookup plugin — most secure and production-grade approach.

Molecule test phases — what happens in each

# verify.yml — your test assertions
# Runs AFTER converge to confirm the role did what it should
# Uses the "assert" module to check conditions
# If any assert fails → molecule test fails → PR cannot merge

- name: Verify nginx role — TechNova test assertions
  hosts: all
  become: true
  gather_facts: true

  tasks:

    # ── Test 1: Nginx package is installed ──────────────────
    - name: Gather installed packages
      package_facts:
        manager: apt
      # package_facts populates ansible_facts.packages dict

    - name: Assert nginx is installed
      assert:
        that:
          - "'nginx' in ansible_facts.packages"
        fail_msg: "FAIL: nginx package is NOT installed!"
        success_msg: "PASS: nginx package is installed ✓"

    # ── Test 2: Nginx service is running ────────────────────
    - name: Gather service facts
      service_facts:
      # Populates ansible_facts.services with all systemd services

    - name: Assert nginx service is running
      assert:
        that:
          - "'nginx.service' in ansible_facts.services"
          - "ansible_facts.services['nginx.service']['state'] == 'running'"
          - "ansible_facts.services['nginx.service']['status'] == 'enabled'"
        fail_msg: "FAIL: nginx service is not running or not enabled!"
        success_msg: "PASS: nginx service is running and enabled ✓"

    # ── Test 3: Config file exists with correct content ──────
    - name: Read nginx config file
      slurp:
        src: /etc/nginx/nginx.conf
      # slurp = reads a file on the remote server into a base64 variable
      register: nginx_conf

    - name: Assert nginx config contains required settings
      assert:
        that:
          - "'worker_processes' in (nginx_conf.content | b64decode)"
          - "'listen 80' in (nginx_conf.content | b64decode)"
        fail_msg: "FAIL: nginx.conf is missing required config!"
        success_msg: "PASS: nginx.conf has correct content ✓"
        # b64decode: decodes the base64 content back to readable text

    # ── Test 4: Port 80 is actually listening ────────────────
    - name: Check if port 80 is open
      wait_for:
        port: 80
        host: localhost
        timeout: 10
        state: started
      # wait_for: waits until the port is open (or fails after timeout)
      # This is stronger than just checking if nginx is running
      # It confirms nginx is actually accepting connections

    # ── Test 5: HTTP response returns 200 ───────────────────
    - name: Verify Nginx returns HTTP 200
      uri:
        url: http://localhost:80
        method: GET
        status_code: 200
        timeout: 10
      register: http_response

    - name: Assert HTTP response is valid
      assert:
        that:
          - "http_response.status == 200"
        fail_msg: "FAIL: Nginx returned HTTP {{ http_response.status }} not 200!"
        success_msg: "PASS: Nginx returns HTTP 200 ✓"

    # ── Test 6: Web root directory exists ────────────────────
    - name: Check web root directory exists
      stat:
        path: /var/www/html
      register: webroot

    - name: Assert web root is a directory with correct permissions
      assert:
        that:
          - webroot.stat.exists
          - webroot.stat.isdir
          - webroot.stat.mode == '0755'
        fail_msg: "FAIL: /var/www/html does not exist or has wrong permissions!"
        success_msg: "PASS: web root directory exists with correct permissions ✓"

🎯 Interview Questions — Chapter 25
Q1: What is Molecule and why do you use it for Ansible roles?
Molecule is a testing framework for Ansible roles that runs roles against Docker containers (or VMs) and verifies the results. It catches bugs before roles are deployed to real servers. Molecule tests: (1) Converge — does the role run without errors? (2) Idempotency — does running twice produce zero changes the second time? (3) Verify — do custom assertions pass (service running, port open, config correct)? Without Molecule, a broken role could run on 200 production servers before anyone notices.
Q2: What is an idempotency test in Molecule and what does a failure mean?
After Molecule applies a role (converge), it runs the role a second time on the same containers. If any task shows "changed" status on this second run, the idempotency test fails. A failure means: the role is not idempotent — it modifies the system every time it runs, even when nothing should change. This is dangerous because nightly scheduled runs would keep modifying servers unnecessarily, potentially causing instability. Common causes: shell tasks that always run (use creates: parameter), file tasks that always write even when content is unchanged, missing changed_when: false on read-only tasks.
Q3: How do you integrate Molecule into a CI/CD pipeline to prevent broken roles from merging?
Create a GitHub Actions workflow (or Jenkinsfile stage) that runs molecule test on every Pull Request that changes role files. Configure the workflow as a required status check in your repository's branch protection rules — PRs cannot be merged until the Molecule workflow passes. Use a matrix strategy to test multiple roles in parallel. The fail-fast: false option shows all failures, not just the first one. This creates an automatic quality gate: Arun's broken log rotation task would cause the idempotency check to fail, preventing the PR from merging, catching the bug before production.

Final Interview Q&A — Senior DevOps Level

🎯 Capstone Interview Questions — Walk Through Your Architecture
Q1: "Walk me through your complete Ansible automation setup at TechNova."
We have a 4-layer automation architecture. Layer 1 is Terraform — all AWS infrastructure (VPC, EC2 ASG, RDS, ALB) across Mumbai and Singapore regions is defined as code with remote S3 state and DynamoDB locking. EC2 instances are tagged Role=webserver and Environment=production. Layer 2 is Ansible — we have 6 roles (common, nginx, java_app, mysql_client, monitoring, docker), all tested with Molecule for idempotency and correctness. Sensitive data is encrypted with Ansible Vault with the password stored in AWS Secrets Manager. Dynamic inventory uses the aws_ec2 plugin to automatically discover servers by EC2 tags. Layer 3 is Jenkins — full CI/CD pipeline: build, test, Docker push to ECR, Ansible deploy to staging with smoke tests, then rolling deploy to production via Tower API. Layer 4 is Ansible Tower — manages all operational automation: Job Templates with RBAC (developers can deploy, auditors can view), scheduled jobs (nightly compliance, weekly patching), PagerDuty alerts on failures, and workflow templates for full release pipelines with approval gates.
Q2: "How do you ensure zero downtime during deployments?"
Zero downtime is achieved through rolling deployment using Ansible's serial: "10%" option. For 200 servers, this processes 20 servers at a time. Each batch follows this sequence: (1) deregister the server from the ALB target group so traffic stops going to it, (2) wait 30 seconds for connection drain so active requests complete, (3) stop the old app, (4) deploy new Docker image, (5) start the new app, (6) health check — retries every 10 seconds for up to 2 minutes, (7) only re-register to ALB after health check passes. If health check fails on any batch, the deployment stops completely — remaining servers keep the old version. Combined with max_fail_percentage: 20, we never lose more than 20% of capacity during a bad deploy.
Q3: "How do you handle secrets in your Ansible setup?"
Three layers of secrets protection: (1) Ansible Vault encrypts the group_vars/all/vault.yml file with AES-256. All sensitive variables (DB passwords, API keys, Slack tokens) have a vault_ prefix and are stored in this encrypted file. Non-secret variables reference vault variables: db_password: "{{ vault_db_password }}". (2) The vault password itself is stored in AWS Secrets Manager. Our ansible.cfg has vault_password_file pointing to a script that retrieves it from Secrets Manager at runtime — the actual password never touches disk. (3) In Ansible Tower, credentials (SSH keys, AWS access keys, vault password) are stored encrypted in Tower's database, injected at runtime. Developers can trigger jobs using these credentials without ever seeing the values. The encrypted vault.yml is safe to commit to Git.
Q4: "How do you prevent broken Ansible roles from reaching production?"
We use Molecule testing in the CI/CD pipeline. Every Pull Request that changes a role file triggers a GitHub Actions workflow that runs molecule test against Docker containers. The test lifecycle covers: (1) YAML lint and ansible-lint for syntax, (2) converge — actually applying the role to containers, (3) idempotency — running the role twice, asserting zero changes on second run, (4) verify — custom assertions checking service status, port availability, config content, HTTP responses. The GitHub repository has branch protection rules requiring all Molecule checks to pass before merging. We test against both Ubuntu 22.04 and 20.04 containers simultaneously. A broken role causes the GitHub Actions check to fail, the PR shows a red X, and it cannot merge to main.
Q5: "How does your dynamic inventory work with AWS Auto Scaling?"
We use the amazon.aws.aws_ec2 inventory plugin configured in aws_ec2.yml. It queries the AWS EC2 DescribeInstances API using boto3 at runtime to get all running instances matching our filters (tag:Project=TechNova, tag:Environment=production). The keyed_groups config automatically creates Ansible groups from EC2 tags — instances tagged Role=webserver join the role_webserver group. When Auto Scaling launches a new instance (tagged correctly by our Launch Template), it appears in Ansible's inventory within 15 minutes (our Tower inventory refresh schedule). When ASG terminates instances, they disappear from inventory automatically. We never manually update any hosts file — the inventory is always live and accurate.
Q6: "What is the most important Ansible lesson you've learned that you'd tell a junior engineer?"
Always test idempotency before considering a playbook production-ready. Run your playbook twice — if any task shows "changed" on the second run, there's a bug. Ansible's idempotency is what makes automation safe to run repeatedly — scheduled jobs, drift correction, CI/CD pipelines — all rely on playbooks being safe to run at any time. Common idempotency violations: shell tasks without creates: parameter, file tasks that rewrite files even when content is identical, service restarts that happen every run instead of only when configuration changed. Fix these with changed_when:, creates: parameters, and most importantly — use handler + notify for restarts instead of always-running restart tasks. Molecule's idempotency test catches all of these before they cause production incidents.
